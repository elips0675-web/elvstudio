    /**
     * json — индекс для клиентского поиска на статической демо-копии.
     */
    public static function json(?array $items = null): string
    {
        $json = json_encode(
            $items ?? static::index(),
            JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES | JSON_HEX_TAG | JSON_HEX_AMP | JSON_HEX_APOS | JSON_HEX_QUOT
        );

        return $json === false ? '[]' : $json;
    }

    /**
     * query — ранжированная выдача для страницы поиска и ответов Консультанта.
     */
    public static function query(string $query, string $type = self::TYPE_ALL, int $limit = 20): array
    {
        $query = static::normalize($query);
        $words = static::words($query);

        if ($query === '' || !$words) {
            return [];
        }

        $results = [];

        foreach (static::index() as $item) {
            if ($type !== '' && $type !== self::TYPE_ALL && ($item['type'] ?? '') !== $type) {
                continue;
            }

            $score = static::score($item, $query, $words);

            if ($score <= 0) {
                continue;
            }

            $item['score'] = $score;
            $results[] = $item;
        }

        usort($results, function (array $a, array $b) {
            if ($a['score'] === $b['score']) {
                return strcmp((string) $a['title'], (string) $b['title']);
            }

            return $b['score'] <=> $a['score'];
        });

        return array_slice($results, 0, max(1, $limit));
    }

    {
        $items = [];

        foreach (Post::query()->published()->with('category')->get() as $post) {

            $tags = implode(' ', $post->tag_list);

                'text' => static::limit(static::plain((string) $post->body), 800),

                'meta' => trim(implode(' · ', array_filter([$category, (string) $post->display_date]))),
            ];
        }

        return $items;
    }

    protected static function score(array $item, string $query, array $words): int
    {
        $title = static::normalize((string) ($item['title'] ?? ''));
        $description = static::normalize((string) ($item['description'] ?? ''));
        $text = static::normalize((string) ($item['text'] ?? ''));
        $keywords = static::normalize((string) ($item['keywords'] ?? ''));
        $meta = static::normalize((string) ($item['meta'] ?? ''));

        $score = $query !== '' && str_contains($title, $query) ? 14 : 0;
        $hits = 0;

        foreach ($words as $word) {
            $forms = static::forms($word);
            $hit = false;

            if (static::containsAny($title, $forms)) {
                $score += 7;
                $hit = true;
            }

            if (static::containsAny($keywords, $forms)) {
                $score += 5;
                $hit = true;
            }

            if (static::containsAny($description, $forms)) {
                $score += 3;
                $hit = true;
            }

            if (static::containsAny($text, $forms)) {
                $score += 2;
                $hit = true;
            }

            if (static::containsAny($meta, $forms)) {
                $score += 1;
                $hit = true;
            }

            if ($hit) {
                $hits++;
            }
        }

        if ($hits === 0) {
            return 0;
        }

        if (count($words) > 1) {
            $score += $hits * 2;
            $score -= (count($words) - $hits) * 2;
        }

        return $score;
    }

    protected static function normalize(string $text): string
    {
        $text = mb_strtolower(strip_tags($text));
        $text = str_replace('ё', 'е', $text);
        $text = (string) preg_replace('/[^\p{L}\p{Nd}\s]+/u', ' ', $text);

        return trim((string) preg_replace('/\s+/u', ' ', $text));
    }

    /**
     * @return string[]
     */
    protected static function words(string $text): array
    {
        $stop = [
            'и', 'в', 'во', 'на', 'для', 'с', 'со', 'по', 'как', 'что', 'это', 'или', 'не',
            'из', 'у', 'о', 'об', 'от', 'до', 'за', 'под', 'над', 'при', 'про', 'без',
            'же', 'бы', 'ли', 'то', 'все', 'мы', 'вы', 'он', 'она', 'они', 'мне', 'нам',
        ];

        $words = array_filter(
            explode(' ', $text),
            function (string $word) use ($stop) {
                return mb_strlen($word) >= 2 && !in_array($word, $stop, true);
            }
        );

        return array_values(array_unique($words));
    }

    /**
     * @return array<int, string>
     */
    protected static function forms(string $word): array
    {
        $forms = [$word];
        $stem = static::stem($word);

        if ($stem !== $word) {
            $forms[] = $stem;
        }

        foreach (static::SYNONYM_GROUPS as $group) {
            if (!in_array($word, $group, true)) {
                continue;
            }

            foreach ($group as $item) {
                $forms[] = $item;
                $stem = static::stem($item);

                if ($stem !== $item) {
                    $forms[] = $stem;
                }
            }

            break;
        }

        return array_values(array_unique(array_filter($forms, function (string $form): bool {
            return mb_strlen($form) >= 2;
        })));
    }

    protected static function stem(string $word): string
    {
        static $endings = [
            'иями', 'ями', 'ами', 'ией', 'иям', 'ием', 'иях', 'ого', 'его', 'ому', 'ему', 'ыми', 'ими',
            'овать', 'евать',
            'ая', 'яя', 'ое', 'ее', 'ые', 'ие', 'ой', 'ей', 'ий', 'ый', 'ом', 'ем', 'ам', 'ям', 'ах', 'ях',
            'ов', 'ев', 'ью', 'ия', 'ию', 'ии', 'ую', 'юю', 'ить', 'ать', 'ять', 'еть',
            'а', 'я', 'о', 'е', 'ы', 'и', 'у', 'ю', 'ь',
        ];

        if (mb_strlen($word) <= 3) {
            return $word;
        }

        foreach ($endings as $ending) {
            $length = mb_strlen($ending);
            $stem = mb_substr($word, 0, -$length);

            if (mb_substr($word, -$length) === $ending && mb_strlen($stem) >= 3) {
                return $stem;
            }
        }

        return $word;
    }

    /**
     * @param array<int, string> $forms
     */
    protected static function containsAny(string $text, array $forms): bool
    {
        foreach ($forms as $form) {
            if (mb_strlen($form) < 4) {
                if (preg_match('/(?<![\p{L}\p{N}])' . preg_quote($form, '/') . '(?![\p{L}\p{N}])/u', $text) === 1) {
                    return true;
                }

                continue;
            }

            if (str_contains($text, $form)) {
                return true;
            }
        }

        return false;
    }

    protected static function plain(string $html): string
    {
        $text = html_entity_decode(strip_tags($html), ENT_QUOTES | ENT_HTML5, 'UTF-8');

        return trim((string) preg_replace('/\s+/u', ' ', $text));
    }

    protected static function limit(string $text, int $length): string
    {
        $text = trim((string) preg_replace('/\s+/u', ' ', $text));

        return mb_strlen($text) > $length ? mb_substr($text, 0, $length) . '…' : $text;
    }
}

        'овать', 'евать',
        'ая', 'яя', 'ое', 'ее', 'ые', 'ие', 'ой', 'ей', 'ий', 'ый', 'ом', 'ем', 'ам', 'ям', 'ах', 'ях',
        'ов', 'ев', 'ью', 'ия', 'ию', 'ии', 'ую', 'юю', 'ить', 'ать', 'ять', 'еть',
        'а', 'я', 'о', 'е', 'ы', 'и', 'у', 'ю', 'ь'
    ];
    var timer = null;

    function normalize(value) {
        var text = String(value || '').toLowerCase().replace(/ё/g, 'е');

        return text.match(/[\p{L}\p{Nd}]+/gu) || [];
    }

    function phrase(value) {
        return normalize(value).join(' ');
    }

    function words(value) {
        var all = normalize(value);
        var seen = {};
        var list = [];

        for (var i = 0; i < all.length; i += 1) {
            var word = all[i];

            if (word.length < 2 || stop[word] || seen[word]) {
                continue;
            }

            seen[word] = true;
            list.push(word);
        }

        return list;
    }

    function stem(word) {
        if (word.length <= 3) {
            return word;
        }

        for (var i = 0; i < endings.length; i += 1) {
            var ending = endings[i];
            var value = word.slice(0, -ending.length);

            if (word.slice(-ending.length) === ending && value.length >= 3) {
                return value;
            }
        }

        return word;
    }

    function formsFor(word) {
        var list = [word];
        var value = stem(word);

        if (value !== word) {
            list.push(value);
        }

        for (var i = 0; i < synonymGroups.length; i += 1) {
            var group = synonymGroups[i];

            if (group.indexOf(word) === -1) {
                continue;
            }

            group.forEach(function (item) {
                list.push(item);

                var itemStem = stem(item);

                if (itemStem !== item) {
                    list.push(itemStem);
                }
            });

            break;
        }

        return list.filter(function (item, index, all) {
            return item.length >= 2 && all.indexOf(item) === index;
        });
    }

    function escapeRegExp(value) {
        return value.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
    }

    function hasForm(text, value) {
        if (value.length < 4) {
            return new RegExp('(^|[^\\p{L}\\p{N}])' + escapeRegExp(value) + '($|[^\\p{L}\\p{N}])', 'u').test(text);
        }

        return text.indexOf(value) !== -1;
    }

    function hasAny(text, forms) {
        for (var i = 0; i < forms.length; i += 1) {
            if (hasForm(text, forms[i])) {
                return true;
            }
        }

        return false;
    }

    function score(item, query, parts) {
        var title = phrase(item.title);
        var description = phrase(item.description);
        var text = phrase(item.text);
        var keywords = phrase(item.keywords);
        var meta = phrase(item.meta);
        var total = title.indexOf(query) !== -1 ? 14 : 0;
        var hits = 0;
        var wordForms = parts.map(formsFor);

        parts.forEach(function (word, index) {
            var forms = wordForms[index];
            var hit = false;

            if (hasAny(title, forms)) {
                total += 7;
                hit = true;
            }

            if (hasAny(keywords, forms)) {
                total += 5;
                hit = true;
            }

            if (hasAny(description, forms)) {
                total += 3;
                hit = true;
            }

            if (hasAny(text, forms)) {
                total += 2;
                hit = true;
            }

            if (hasAny(meta, forms)) {
                total += 1;
                hit = true;
            }

            if (hit) {
                hits += 1;
            }
        });

        if (hits === 0) {
            return 0;
        }

        if (parts.length > 1) {
            total += hits * 2;
            total -= (parts.length - hits) * 2;
        }

        return total;
    }

    function plural(count, one, few, many) {
        var last = count % 10;
        var lastTwo = count % 100;

        if (last === 1 && lastTwo !== 11) {
            return one;
        }

        if (last >= 2 && last <= 4 && (lastTwo < 10 || lastTwo >= 20)) {
            return few;
        }

        return many;
    }

    function appendHighlighted(node, value, parts) {
        var source = String(value || '');
        var forms = [];

        parts.forEach(function (word) {
            forms = forms.concat(formsFor(word));
        });

        forms = forms.filter(function (form, index, all) {
            return form.length >= 4 && all.indexOf(form) === index;
        });

        if (!forms.length) {
            node.textContent = source;
            return;
        }

        forms.sort(function (a, b) {
            return b.length - a.length;
        });

        var pattern = new RegExp('(' + forms.map(escapeRegExp).join('|') + ')', 'giu');
        var last = 0;
        var match;

        while ((match = pattern.exec(source)) !== null) {
            if (match.index > last) {
                node.appendChild(document.createTextNode(source.slice(last, match.index)));
            }

            var mark = document.createElement('mark');
            mark.textContent = match[0];
            node.appendChild(mark);
            last = match.index + match[0].length;

            if (match[0] === '') {
                pattern.lastIndex += 1;
            }
        }

        if (last < source.length) {
            node.appendChild(document.createTextNode(source.slice(last)));
        }
    }

    function card(item, parts) {
        var link = document.createElement('a');
        link.className = 'search-result';
        link.href = item.url;

        var top = document.createElement('div');
        top.className = 'search-result__top';

        var type = document.createElement('span');
        type.className = 'search-result__type';
        type.textContent = item.type === 'blog' ? 'Статья' : 'Страница';
        top.appendChild(type);

        if (item.group) {
            var group = document.createElement('span');
            group.className = 'search-result__group';
            group.textContent = item.group;
            top.appendChild(group);
        }

        link.appendChild(top);

        var title = document.createElement('h2');
        appendHighlighted(title, item.title || '', parts);
        link.appendChild(title);

        var description = item.description || item.text || '';
        if (description) {
            var lead = document.createElement('p');
            appendHighlighted(lead, description, parts);
            link.appendChild(lead);
        }

        if (item.meta) {
            var meta = document.createElement('span');
            meta.className = 'search-result__meta';
            meta.textContent = item.meta;
            link.appendChild(meta);
        }

        return link;
    }

    function render() {
        if (!loaded) {
            statusNode.textContent = 'Загружаем поиск…';
            return;
        }

        var raw = input.value.trim();
        var query = phrase(raw);
        var parts = words(raw);

        results.textContent = '';
        results.setAttribute('aria-busy', 'false');
        emptyNode.hidden = true;

        if (!raw) {
            popularNode.hidden = false;

            if (moreNode) {
                moreNode.hidden = true;
            }

            statusNode.textContent = 'Введите запрос — покажем страницы и статьи.';
            return;
        }

        popularNode.hidden = true;
        results.setAttribute('aria-busy', 'true');

        if (!parts.length) {
            if (moreNode) {
                moreNode.hidden = true;
            }

            results.setAttribute('aria-busy', 'false');
            statusNode.textContent = 'Введите хотя бы два символа.';
            return;
        }

        var found = [];

        items.forEach(function (item) {
            if (state.type !== 'all' && item.type !== state.type) {
                return;
            }

            var value = score(item, query, parts);

            if (value > 0) {
                found.push({ item: item, score: value });
            }
        });

        found.sort(function (a, b) {
            if (a.score === b.score) {
                return String(a.item.title).localeCompare(String(b.item.title), 'ru');
            }

            return b.score - a.score;
        });

        if (!found.length) {
            if (moreNode) {
                moreNode.hidden = true;
            }

            results.setAttribute('aria-busy', 'false');
            emptyNode.hidden = false;
            statusNode.textContent = 'Ничего не нашлось по запросу «' + raw + '».';
            return;
        }

        var shown = found.slice(0, state.limit);
        var suffix = plural(found.length, 'результат', 'результата', 'результатов');
        statusNode.textContent = found.length > shown.length
            ? 'Нашли ' + found.length + ' ' + suffix + ', показываем первые ' + shown.length + '.'
            : 'Нашли ' + found.length + ' ' + suffix + '.';

        shown.forEach(function (entry) {
            results.appendChild(card(entry.item, parts));
        });

        if (moreNode) {
            moreNode.hidden = found.length <= shown.length;
        }

        results.setAttribute('aria-busy', 'false');
    }

    function syncUrl() {
        var value = input.value.trim();
        var base = window.location.pathname.replace(/index\.html$/, '');

        if (base.charAt(base.length - 1) !== '/') {
            base += '/';
        }

        try {
            window.history.replaceState(null, '', base + (value ? '?q=' + encodeURIComponent(value) : ''));
        } catch (error) {
            /* file:// не даёт менять адрес — поиск всё равно работает. */
        }
    }

    input.addEventListener('input', function () {
        state.limit = 30;
        window.clearTimeout(timer);
        timer = window.setTimeout(function () {
            syncUrl();
            render();
        }, 120);
    });

    form.addEventListener('submit', function (event) {
        event.preventDefault();
        state.limit = 30;
        syncUrl();
        render();
    });

    Array.prototype.forEach.call(filterNodes, function (button) {
        button.addEventListener('click', function () {
            state.type = button.getAttribute('data-search-type') || 'all';
            state.limit = 30;

            Array.prototype.forEach.call(filterNodes, function (node) {
                node.classList.toggle('chip--active', node === button);
            });

            render();
        });
    });

    if (moreNode) {
        moreNode.addEventListener('click', function () {
            state.limit += 30;
            render();
        });
    }

    statusNode.textContent = 'Загружаем поиск…';

    var initial = new URLSearchParams(window.location.search).get('q');

    if (initial && !input.value) {
        input.value = initial;
    }

    fetch(indexNode.getAttribute('data-search-index') || '/search-index.json', {
        headers: { Accept: 'application/json' }
    })
        .then(function (response) {
            if (!response.ok) {
                throw new Error('search index ' + response.status);
            }

            return response.json();
        })
        .then(function (data) {
            items = Array.isArray(data) ? data : [];
            loaded = true;

            if (input.value.trim()) {
                render();
                return;
            }

            statusNode.textContent = 'Введите запрос — покажем страницы и статьи.';
            popularNode.hidden = false;
        })
        .catch(function () {
            statusNode.textContent = 'Поиск временно недоступен. Откройте карту сайта.';
        });
})();
