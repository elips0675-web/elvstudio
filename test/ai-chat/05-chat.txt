    protected static function reply(string $answer, array $chips = [], ?array $link = null, ?string $open = null, array $sources = [], bool $rejected = false): array
    {
        return [
            'answer' => $answer,
            'chips' => array_values(array_filter($chips)),
            'link' => $link,
            'open' => $open,
            'sources' => array_values($sources),
            'rejected' => $rejected,
        ];
    }

    /**
     * isNonsense — не пропускает одиночный набор букв и текст без знакомых слов.
     */
    protected static function isNonsense(string $text): bool
    {
        $words = preg_split('/\s+/u', trim($text), -1, PREG_SPLIT_NO_EMPTY) ?: [];

        foreach ($words as $word) {
            if (mb_strlen($word) >= 3 && Search::query($word, Search::TYPE_ALL, 1) !== []) {
                return false;
            }
        }

        return true;
    }

    /**
     * search — подбирает страницы и статьи по вопросу пользователя.
     */
    protected static function search(string $message): array
    {
        return Search::query($message, Search::TYPE_ALL, 3);
    }

    /**
     * searchReply — отвечает ссылками на найденные материалы, не пересказывая их.
     */
    protected static function searchReply(array $items, bool $fallback = false): array
    {
        $sources = [];
        $seen = [];

        foreach ($items as $item) {
            $label = trim((string) ($item['title'] ?? ''));
            $path = trim((string) ($item['url'] ?? ''));

            if ($label === '' || $path === '' || isset($seen[$path])) {
                continue;
            }

            $seen[$path] = true;
            $sources[] = [
                'label' => $label,
                'url' => rtrim(url($path), '/') . '/',
                'kind' => ($item['type'] ?? '') === Search::TYPE_BLOG ? 'Статья' : 'Страница',
            ];
        }

        if ($sources === []) {
            return static::reply(
                'По этому запросу на сайте ничего не нашёл. Попробуйте сформулировать короче или напишите нам — подскажем вручную.',
                ['Сколько стоит?', 'Какие разделы есть?', 'Написать нам'],
                null,
                'feedback'
            );
        }

        return static::reply(
            ($fallback ? 'Точного ответа нет, но на сайте есть материалы по теме.' : 'Нашёл материалы по теме.') . ' Откройте нужную ссылку ниже или уточните вопрос.',
            ['Сколько стоит?', 'Что входит в Kit?', 'Написать нам'],
            null,
            null,
            $sources
        );
    }

    protected static function link(string $label, string $path): array
    {
        return ['label' => $label, 'url' => url($path)];
    }

    protected static function normalize(string $text): string
    {
        $text = mb_strtolower(strip_tags($text));
        $text = str_replace('ё', 'е', $text);
        $text = (string) preg_replace('/[^\p{L}\p{Nd}\s]+/u', ' ', $text);

        return trim((string) preg_replace('/\s+/u', ' ', $text));
    }

    protected static function has(string $text, array $words): bool
    {
        foreach ($words as $word) {
            if ($word !== '' && mb_strpos($text, $word) !== false) {
                return true;
            }
        }

        return false;
    }

    protected static function plansText(): string
    {
        $plans = Catalog::plans();

        if (!$plans) {
            return 'Каталог пока не заполнен. Напишите нам — пришлём тарифы.';
        }

        $lines = ['Тарифы Elv Kit:'];

        foreach ($plans as $plan) {
            $lines[] = '• ' . $plan['label'] . ' — ' . Money::format((int) $plan['price']) . ': ' . $plan['period'] . '.';
        }

        $lines[] = 'Оплата разовая, без подписки.';

        return implode("\n", $lines);
    }

    protected static function modulesText(): string
    {
        $modules = Catalog::modules();

        if (!count($modules)) {
            return 'Разделы: онлайн-запись, заказы и оплата, обратная связь продаются отдельно, блог входит в Elv Kit.';
        }

        $notes = [
            'booking' => 'онлайн-запись, занятое время исчезает',
            'shop' => 'заказы, оплата и выдача файла',
            'feedback' => 'форма и виджет обратной связи',

        ];

        $lines = ['Разделы продаются отдельно:'];

        foreach ($modules as $module) {
            $note = $notes[$module->slug] ?? '';
            $lines[] = '• ' . $module->name . ' — ' . $module->price_label . ($note !== '' ? ': ' . $note : '');

        $lines[] = 'Блог входит в Elv Kit.';

    protected static function modulePriceText(string $text): string
    {
        $slug = null;

        foreach (['booking' => 'booking', 'shop' => 'shop', 'feedback' => 'feedback', 'блог' => 'blog', 'blog' => 'blog', 'запис' => 'booking', 'заказ' => 'shop', 'магазин' => 'shop', 'обратн' => 'feedback', 'связ' => 'feedback'] as $word => $key) {
            if (static::has($text, [$word])) {
                $slug = $key;
                break;
            }
        }

        if ($slug === null) {
            return static::modulesText();
        }

        foreach (Catalog::modules() as $module) {
            if ($module->slug === $slug) {
                return $module->name . ' — ' . $module->price_label . '. Продаётся отдельно от Elv Kit и подключается к сайту.';
            }
        }

        return static::modulesText();
    }

    protected static function servicesText(): string
    {
        $services = Slots::services();

        if (!count($services)) {
            return '';
        }

        $lines = ["\nВ демо-записи:"];

        foreach ($services as $service) {
            $lines[] = '• ' . $service['name'] . ' — ' . $service['duration'] . ' мин, ' . $service['price'] . '.';
        }

        return implode("\n", $lines);
    }

    protected static function contactsText(): string
    {
        $parts = [];

        $email = (string) Settings::get('email');
        $phone = (string) Settings::get('phone');
        $telegram = (string) Settings::get('telegram');

        if ($email !== '') {
            $parts[] = 'E-mail: ' . $email;
        }

        if ($phone !== '') {
            $parts[] = 'Телефон: ' . $phone;
        }

        if ($telegram !== '') {
            $parts[] = 'Telegram: ' . $telegram;
        }

        if (!$parts) {
            return 'Оставьте сообщение в форме обратной связи — ответим на e-mail.';
        }

        return implode("\n", $parts) . "\nИли оставьте сообщение в форме обратной связи.";
    }

<?php namespace Elv\Kit\Classes;

/**
 * TextFilter — фильтр грубых слов в сообщениях консультанта.
 *
 * Ловит основы слов, а не конкретные формы: между буквами допускаются пробелы
 * и знаки (`х у й`), повтор букв (`пиззда`), латинские похожие буквы (`xyй`)
 * и замена буквы знаком (`х*й`). Слово, где букву просто выбросили (`хй`),
 * фильтр не разгадывает — иначе подстановка любых символов ловит обычные слова.
 * Ответ на грубость вежливый, без повтора самого слова.
 */
class TextFilter
{
    public const REFUSAL = 'Давайте без грубостей — отвечу по делу. Спросите про тарифы, разделы сайта, установку, оплату или ключи.';

    /** @var array<string, string> */
    protected const LOOKALIKE = [
        'a' => 'а', 'b' => 'в', 'c' => 'с', 'e' => 'е', 'h' => 'н', 'k' => 'к',
        'm' => 'м', 'o' => 'о', 'p' => 'р', 't' => 'т', 'x' => 'х', 'y' => 'у',
    ];

    /** @var array<string, string> */
    protected const REVERSE = [
        'а' => 'a', 'в' => 'b', 'с' => 'c', 'е' => 'e', 'н' => 'h', 'к' => 'k',
        'м' => 'm', 'о' => 'o', 'р' => 'p', 'т' => 't', 'х' => 'x', 'у' => 'y',
    ];

    /** @var array<string, string> */
    protected static $compiled = [];

    public static function has(string $text): bool
    {
        return static::match($text) !== null;
    }

    public static function match(string $text): ?string
    {
        $text = mb_strtolower($text);
        $text = str_replace('ё', 'е', $text);

        if (trim($text) === '') {
            return null;
        }

        $latin = strtr($text, static::REVERSE);

        foreach (static::latinWords() as $stem) {
            if (preg_match(static::pattern($stem), $latin)) {
                return $stem;
            }
        }

        $cyrillic = strtr($text, static::LOOKALIKE);

        foreach (static::cyrillicWords() as $stem) {
            if (preg_match(static::pattern($stem), $cyrillic)) {
                return $stem;
            }
        }

        return null;
    }

    /**
     * @return array<int, string>
     */
    protected static function cyrillicWords(): array
    {
        return [
            'хуй', 'хуя', 'хую', 'хуе', 'хуи',
            'пизд',
            'бляд', 'блят',
            'ебат', 'ебал', 'ебан', 'ебет', 'ебуч',
            'заеб', 'наеб', 'уеб', 'отъеб', 'выеб', 'доеб', 'проеб', 'съеб',
            'мудак', 'мудил',
            'пидор', 'пидар', 'пидр',
            'гандон', 'гондон',
            'шлюх',
            'долбоеб',
            'залуп',
            'дроч',
            'мразь', 'мрази',
            'гнида',
        ];
    }

    /**
     * @return array<int, string>
     */
    protected static function latinWords(): array
    {
        return [
            'fuck', 'shit', 'bitch', 'cunt', 'asshole', 'bastard', 'dick', 'pussy', 'whore', 'slut',
        ];
    }

    /**
     * pattern — буквы основы через любые пробелы и знаки, повторы букв разрешены.
     */
    protected static function pattern(string $stem): string
    {
        if (isset(static::$compiled[$stem])) {
            return static::$compiled[$stem];
        }

        $letters = preg_split('//u', $stem, -1, PREG_SPLIT_NO_EMPTY) ?: [];
        $parts = [];
        $last = count($letters) - 1;

        foreach ($letters as $index => $letter) {
            $quoted = preg_quote($letter, '/');

            if ($index === 0 || $index === $last) {
                $parts[] = $quoted . '+';
                continue;
            }

            $parts[] = '(?:' . $quoted . '+|[\p{P}\p{S}]+)';
        }

        return static::$compiled[$stem] = '/' . implode('[\s\p{P}\p{S}]*', $parts) . '/u';
    }
}

<?php namespace Elv\Kit\Classes\Http;

use Elv\Kit\Classes\AntiSpam;
use Elv\Kit\Classes\Chat;
use Illuminate\Http\Request;
use Illuminate\Routing\Controller;
use Illuminate\Support\Facades\Validator;

/**
 * ChatController — ответы Консультанта по базе знаний сайта.
 *
 * Внешних API нет: ответ собирается из каталога и настроек, поэтому работает
 * и на демо-контуре без ключей. Лимит запросов задан в routes.php.
 */
class ChatController extends Controller
{
    public function store(Request $request)
    {
        if (AntiSpam::isBot($request)) {
            return AntiSpam::json([
                'ok' => true,
                'answer' => 'Здравствуйте! Чем помочь по Elv Kit?',
                'chips' => ['Сколько стоит?', 'Как установить?'],
                'link' => null,
                'open' => null,
                'sources' => [],
                'rejected' => false,
            ]);
        }

        $validator = Validator::make($request->all(), [
            'message' => 'required|string|min:2|max:' . Chat::MAX_LENGTH,
        ], [
            'message.required' => 'Напишите вопрос.',
            'message.min' => 'Напишите вопрос подробнее — от двух символов.',
            'message.max' => 'Слишком длинный вопрос: до ' . Chat::MAX_LENGTH . ' символов.',
        ]);

        if ($validator->fails()) {
            return AntiSpam::json([
                'ok' => false,
                'code' => 'VALIDATION',
                'errors' => $validator->errors()->toArray(),
            ], 422);
        }

        $result = Chat::answer((string) $request->input('message'));

        return AntiSpam::json(['ok' => true] + $result);
    }
}

(function () {
    'use strict';

    var doc = document;
    var csrfMeta = doc.querySelector('meta[name="csrf-token"]');
    var body = doc.body;

    function csrf() {
        return csrfMeta ? csrfMeta.getAttribute('content') : '';
    }

    var fab = doc.querySelector('[data-fab]');
    var feedback = doc.querySelector('[data-feedback]');
    var chat = doc.getElementById('elv-chat');
    var openFeedback = null;
    var openChat = null;
    var closeChat = null;

    if (feedback) {
        var resetFeedback = function () {
            var form = feedback.querySelector('form');
            if (!form) {
                return;
            }
            Array.prototype.forEach.call(feedback.querySelectorAll('.alert'), function (box) {
                box.remove();
            });
            form.classList.remove('is-hidden');
        };

        openFeedback = function () {
            resetFeedback();
            if (typeof feedback.showModal === 'function') {
                if (!feedback.open) {
                    feedback.showModal();
                }
            }
            else {
                feedback.setAttribute('open', 'open');
            }

            var first = feedback.querySelector('input[name="name"]');
            if (first) {
                first.focus();
            }
        };

        var closeFeedback = function () {
            if (typeof feedback.close === 'function') {
                feedback.close();
            }
            else {
                feedback.removeAttribute('open');
            }
        };

        Array.prototype.forEach.call(doc.querySelectorAll('[data-feedback-open]'), function (button) {
            button.addEventListener('click', function (event) {
                event.preventDefault();
                if (typeof closeChat === 'function') {
                    closeChat();
                }
                openFeedback();
            });
        });

        Array.prototype.forEach.call(feedback.querySelectorAll('[data-feedback-close]'), function (button) {
            button.addEventListener('click', closeFeedback);
        });

        feedback.addEventListener('click', function (event) {
            if (event.target === feedback) {
                closeFeedback();
            }
        });

        feedback.addEventListener('close', resetFeedback);

        feedback.addEventListener('elv:form-sent', function () {
            window.setTimeout(function () {
                if (feedback.open) {
                    closeFeedback();
                }
            }, 2200);
        });
    }

    if (chat) {
        var chatLog = doc.getElementById('elv-chat-log');
        var chatForm = chat.querySelector('[data-chat-form]');
        var chatInput = doc.getElementById('elv-chat-input');
        var chatStatus = chat.querySelector('[data-chat-status]');
        var chatSuggest = chat.querySelector('[data-chat-suggest]');
        var chatError = chat.querySelector('[data-chat-error]');
        var chatSend = chat.querySelector('.elv-chat__send');
        var chatExpand = chat.querySelector('[data-chat-expand]');
        var chatOpeners = doc.querySelectorAll('[data-chat-open]');
        var chatPrompt = doc.querySelector('[data-chat-prompt]');
        var promptKey = 'elvChatPrompt';
        var busy = false;
        var greeted = false;

        var setOpeners = function (open) {
            Array.prototype.forEach.call(chatOpeners, function (button) {
                button.setAttribute('aria-expanded', open ? 'true' : 'false');
            });
        };

        var bubble = function (who, text) {
            var row = doc.createElement('div');
            row.className = 'elv-chat__msg elv-chat__msg--' + who;
            var box = doc.createElement('div');
            box.className = 'elv-chat__bubble';
            box.textContent = text;
            row.appendChild(box);
            chatLog.appendChild(row);
            chatLog.scrollTop = chatLog.scrollHeight;

            return { row: row, box: box };
        };

        var addLink = function (link) {
            if (!link || !link.url || !link.label) {
                return;
            }

            var anchor = doc.createElement('a');
            anchor.className = 'elv-chat__link';
            anchor.href = link.url;
            anchor.textContent = link.label;
            chatLog.appendChild(anchor);
            chatLog.scrollTop = chatLog.scrollHeight;
        };

        var setChips = function (chips) {
            if (!chatSuggest || !chips || !chips.length) {
                return;
            }

            chatSuggest.textContent = '';

            chips.forEach(function (text) {
                var chip = doc.createElement('button');
                chip.className = 'elv-chat__chip';
                chip.type = 'button';
                chip.textContent = text;
                chip.addEventListener('click', function () {
                    ask(text);
                });
                chatSuggest.appendChild(chip);
            });
        };

        var showError = function (text) {
            if (!chatError) {
                return;
            }

            chatError.textContent = text;
            chatError.hidden = !text;
        };

        var setBusy = function (state) {
            busy = state;
            if (chatSend) {
                chatSend.disabled = state;
            }
            if (chatStatus) {
                chatStatus.textContent = state ? 'Печатает…' : 'Отвечает по базе знаний сайта';
            }
        };

        var ask = function (text) {
            var message = String(text || '').trim();

            if (!message || busy) {
                return;
            }

            if (message.length > 500) {
                message = message.slice(0, 500);
            }

            showError('');
            var user = bubble('user', message);

            if (chatInput) {
                chatInput.value = '';
            }

            var typing = bubble('bot', '');
            typing.box.className = 'elv-chat__bubble elv-chat__typing';
            typing.box.innerHTML = '<i></i><i></i><i></i>';
            setBusy(true);

            fetch('/chat', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Accept': 'application/json',
                    'X-Requested-With': 'XMLHttpRequest',
                    'X-CSRF-TOKEN': csrf()
                },
                body: JSON.stringify({ message: message })
            })
                .then(function (response) {
                    return response.json().catch(function () {
                        return { ok: false };
                    }).then(function (data) {
                        return { status: response.status, data: data };
                    });
                })
                .then(function (result) {
                    typing.row.remove();

                    if (result.status === 200 && result.data && result.data.ok) {
                        if (result.data.rejected) user.row.remove();
                        bubble('bot', result.data.answer || '');

                        if (result.data.sources) result.data.sources.forEach(addLink);
                        addLink(result.data.link);

                        if (result.data.open === 'feedback' && typeof openFeedback === 'function') {
                            var button = doc.createElement('button');
                            button.className = 'elv-chat__link';
                            button.type = 'button';
                            button.textContent = 'Открыть форму обратной связи';
                            button.addEventListener('click', function () {
                                closeChat();
                                openFeedback();
                            });
                            chatLog.appendChild(button);
                            chatLog.scrollTop = chatLog.scrollHeight;
                        }

                        setChips(result.data.chips);
                    }
                    else if (result.status === 429) {
                        showError('Слишком много сообщений подряд. Подождите минуту.');
                    }
                    else if (result.status === 422) {
                        showError('Напишите вопрос подробнее — от двух символов.');
                    }
                    else {
                        showError('Консультант недоступен. Напишите нам через форму обратной связи.');
                    }
                })
                .catch(function () {
                    typing.row.remove();
                    showError('Сеть недоступна. Попробуйте ещё раз.');
                })
                .then(function () {
                    setBusy(false);

                    if (chatInput) {
                        chatInput.focus();
                    }
                });
        };

        openChat = function () {
            if (!chat.hidden) {
                if (chatInput) {
                    chatInput.focus();
                }
                return;
            }

            chat.hidden = false;
            body.classList.add('is-chat-open');
            setOpeners(true);

            if (!greeted) {
                greeted = true;
                bubble('bot', chatLog.getAttribute('data-greeting') || '');
            }

            if (chatPrompt) {
                chatPrompt.hidden = true;
            }

            if (chatInput) {
                chatInput.focus();
            }
        };

        closeChat = function () {
            if (chat.hidden) {
                return;
            }

            chat.hidden = true;
            body.classList.remove('is-chat-open');
            setOpeners(false);
        };

        Array.prototype.forEach.call(chatOpeners, function (button) {
            button.addEventListener('click', function (event) {
                event.preventDefault();
                openChat();
            });
        });

        Array.prototype.forEach.call(chat.querySelectorAll('[data-chat-close]'), function (button) {
            button.addEventListener('click', closeChat);
        });

        if (chatForm) {
            chatForm.addEventListener('submit', function (event) {
                event.preventDefault();
                ask(chatInput ? chatInput.value : '');
            });
        }

        if (chatSuggest) {
            Array.prototype.forEach.call(chatSuggest.querySelectorAll('[data-chat-q]'), function (chip) {
                chip.addEventListener('click', function () {
                    ask(chip.getAttribute('data-chat-q'));
                });
            });
        }

        if (chatExpand) {
            chatExpand.addEventListener('click', function () {
                var expanded = chat.classList.toggle('is-expanded');
                chatExpand.setAttribute('aria-pressed', expanded ? 'true' : 'false');
            });
        }

        doc.addEventListener('keydown', function (event) {
            if (event.key === 'Escape' && !chat.hidden) {
                closeChat();
            }
        });

        if (chatPrompt) {
            var seen = false;

            try {
                seen = window.sessionStorage.getItem(promptKey) === '1';
            }
            catch (error) {
                seen = true;
            }

            if (!seen) {
                window.setTimeout(function () {
                    if (chat.hidden) {
                        chatPrompt.hidden = false;
                    }
                }, 8000);
            }

            var hidePrompt = function () {
                chatPrompt.hidden = true;
                try {
                    window.sessionStorage.setItem(promptKey, '1');
                }
                catch (error) {
                    return;
                }
            };

            var promptClose = chatPrompt.querySelector('[data-chat-prompt-close]');
            if (promptClose) {
                promptClose.addEventListener('click', hidePrompt);
            }
        }
    }

    if (fab) {
        var fabToggle = fab.querySelector('[data-fab-toggle]');
        var fabLinks = Number(fabToggle ? fabToggle.getAttribute('data-links') : 0) || 0;

        var setFab = function (open) {
            fab.classList.toggle('is-open', open);
            if (fabToggle) {
                fabToggle.setAttribute('aria-expanded', open ? 'true' : 'false');
            }
        };

        if (fabToggle) {
            fabToggle.addEventListener('click', function (event) {
                event.stopPropagation();

                if (fabLinks === 0) {
                    setFab(false);

                    if (typeof openChat === 'function' && chat) {
                        openChat();
                    }
                    else if (typeof openFeedback === 'function') {
                        openFeedback();
                    }

                    return;
                }

                setFab(!fab.classList.contains('is-open'));
            });
        }

        Array.prototype.forEach.call(fab.querySelectorAll('.fab__item'), function (item) {
            item.addEventListener('click', function () {
                setFab(false);
            });
        });

        doc.addEventListener('click', function (event) {
            if (!fab.contains(event.target)) {
                setFab(false);
            }
        });

        doc.addEventListener('keydown', function (event) {
            if (event.key === 'Escape') {
                setFab(false);
            }
        });
    }

    var toTop = doc.querySelector('[data-to-top]');

    if (toTop) {
        var onScroll = function () {
            toTop.classList.toggle('is-visible', window.pageYOffset > 600);
        };

        window.addEventListener('scroll', onScroll, { passive: true });
        onScroll();

        toTop.addEventListener('click', function () {
            window.scrollTo({ top: 0, behavior: 'smooth' });
        });
    }
})();
