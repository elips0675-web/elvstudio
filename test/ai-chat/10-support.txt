<?php namespace Elv\Kit\Classes;

use Carbon\Carbon;

/**
 * Support — рабочее время поддержки и срок первого ответа.
 *
 * SLA считается по рабочим часам, а не по календарным суткам: обращение,
 * пришедшее в пятницу вечером, не должно «просрочиваться» за выходные.
 * Рабочее время — Пн–Пт 10:00–18:00 Europe/Minsk, срок — 24 рабочих часа.
 */
class Support
{
    public const TIMEZONE = 'Europe/Minsk';
    public const WORK_START = 10;
    public const WORK_END = 18;
    public const SLA_HOURS = 24;

    /**
     * dueAt — срок первого ответа от момента создания обращения.
     */
    public static function dueAt($start = null): Carbon
    {
        // Строка трактуется как местное время поддержки: `now()` приходит
        // в UTC приложения, а «пятница 17:30» в коде и тестах — это Минск.
        if ($start instanceof \DateTimeInterface) {
            $time = Carbon::instance($start)->setTimezone(self::TIMEZONE);
        }
        elseif ($start) {
            $time = Carbon::parse((string) $start, self::TIMEZONE);
        }
        else {
            $time = now()->setTimezone(self::TIMEZONE);
        }

        $remaining = self::SLA_HOURS * 3600;

        while ($remaining > 0) {
            $time = self::nextWorkingMoment($time);

            $endOfDay = $time->copy()->setTime(self::WORK_END, 0, 0);
            $available = $endOfDay->getTimestamp() - $time->getTimestamp();

            if ($available <= 0) {
                $time = self::nextWorkingMoment($endOfDay->copy()->addMinute());
                continue;
            }

            if ($available >= $remaining) {
                $time = $time->copy()->addSeconds($remaining);
                $remaining = 0;
                continue;
            }

            $remaining -= $available;
            $time = self::nextWorkingMoment($endOfDay);
        }

        return $time->setTimezone((string) config('app.timezone', 'UTC'));
    }

    /**
     * nextWorkingMoment — ближайшая минута внутри рабочего окна.
     */
    public static function nextWorkingMoment(Carbon $time): Carbon
    {
        $time = $time->copy()->setTimezone(self::TIMEZONE);

        while (true) {
            if (!self::isWorkday($time)) {
                $time = $time->copy()->addDay()->setTime(self::WORK_START, 0, 0);
                continue;
            }

            if ($time->hour < self::WORK_START) {
                return $time->setTime(self::WORK_START, 0, 0);
            }

            if ($time->hour >= self::WORK_END) {
                $time = $time->copy()->addDay()->setTime(self::WORK_START, 0, 0);
                continue;
            }

            return $time;
        }
    }

    public static function isWorkday(Carbon $time): bool
    {
        return $time->dayOfWeekIso >= 1 && $time->dayOfWeekIso <= 5;
    }

    public static function isWorkingMoment(Carbon $time): bool
    {
        $time = $time->copy()->setTimezone(self::TIMEZONE);

        return self::isWorkday($time)
            && $time->hour >= self::WORK_START
            && $time->hour < self::WORK_END;
    }

    /**
     * label — срок в часовом поясе поддержки, без машинного UTC.
     */
    public static function label($time): string
    {
        if (!$time instanceof Carbon) {
            return '';
        }

        return $time->copy()->setTimezone(self::TIMEZONE)->format('d.m.Y H:i');
    }

    /**
     * isOverdue — первый ответ не отправлен, а срок уже прошёл.
     */
    public static function isOverdue($ticket): bool
    {
        if (!$ticket || !$ticket->sla_due_at instanceof Carbon) {
            return false;
        }

        if ($ticket->first_response_at instanceof Carbon) {
            return false;
        }

        if ((string) $ticket->status === 'closed') {
            return false;
        }

        return $ticket->sla_due_at->isPast();
    }

    /**
     * describe — человеческая строка для писем и админки.
     */
    public static function describe($ticket): string
    {
        if (!$ticket || !$ticket->sla_due_at instanceof Carbon) {
            return '';
        }

        if ($ticket->first_response_at instanceof Carbon) {
            return 'Первый ответ: ' . self::label($ticket->first_response_at);
        }

        return 'Первый ответ до: ' . self::label($ticket->sla_due_at);
    }
}

<?php namespace Elv\Kit\Classes;

use Elv\Kit\Classes\Mail\TicketAdmin;
use Elv\Kit\Classes\Mail\TicketCustomer;
use Elv\Kit\Models\Customer;
use Elv\Kit\Models\Settings;
use Elv\Kit\Models\Ticket;
use Elv\Kit\Models\TicketMessage;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\Mail;

/**
 * TicketService — создание обращений, ответы и уведомления.
 *
 * Обращение сначала сохраняется, потом ставятся письма и Telegram: если
 * внешний канал лежит, история остаётся в базе и видна в кабинете и админке.
 *
 * Три входа в поддержку дают одно обращение: кабинет, форма обратной связи и
 * кнопка «Передать оператору» в Консультанте. У формы и чата покупатель может
 * быть без входа — клиент создаётся по e-mail, чтобы обращение было видно в
 * кабинете после входа по Magic Link.
 */
class TicketService
{
    /**
     * open — обращение из кабинета: покупатель уже известен.
     *
     * @param  array{subject: string, body: string, order_id?: int|null, attachments?: array<int, array<string, mixed>>}  $data
     */
    public function open(Customer $customer, array $data): Ticket
    {
        return $this->create([
            'customer_id' => (int) $customer->id,
            'email' => (string) $customer->email,
            'name' => (string) $customer->display_name,
            'subject' => (string) $data['subject'],
            'body' => (string) $data['body'],
            'order_id' => !empty($data['order_id']) ? (int) $data['order_id'] : null,
            'source' => (string) ($data['source'] ?? 'account'),
            'attachments' => (array) ($data['attachments'] ?? []),
        ]);
    }

    /**
     * openGuest — обращение из формы обратной связи без входа.
     *
     * Клиент создаётся или находится по e-mail: покупатель входит по Magic Link
     * на тот же адрес и видит обращение в кабинете. Существующая карточка не
     * перезаписывается, пустое имя заполняется из формы.
     *
     * @param  array{email: string, name?: string, subject?: string, body: string, order_id?: int|null, attachments?: array<int, array<string, mixed>>}  $data
     */
    public function openGuest(array $data): Ticket
    {
        $email = mb_strtolower(trim((string) $data['email']));
        $name = trim((string) ($data['name'] ?? ''));

        // Обращение с формы должно быть видно в кабинете: покупатель входит по
        // Magic Link на этот же e-mail. Поэтому клиент создаётся, а не только
        // ищется; существующая карточка не перезаписывается.
        $customer = Customer::query()->firstOrCreate(['email' => $email], ['name' => $name]);

        if ($name !== '' && trim((string) $customer->name) === '') {
            $customer->forceFill(['name' => $name])->save();
        }

        return $this->create([
            'customer_id' => (int) $customer->id,
            'email' => $email,
            'name' => $name,
            'subject' => (string) ($data['subject'] ?? ''),
            'body' => (string) $data['body'],
            'order_id' => !empty($data['order_id']) ? (int) $data['order_id'] : null,
            'source' => 'feedback',
            'attachments' => (array) ($data['attachments'] ?? []),
        ]);
    }

    /**
     * handoff — обращение из Консультанта с историей переписки.
     *
     * История приходит списком `role` (`user`/`bot`) и `text`: ответы бота
     * сохраняются отдельным автором, чтобы поддержка видела, что уже сказал
     * консультант, и не отвечала то же самое.
     *
     * @param  array{email: string, name?: string, message?: string, history?: array<int, array{role?: string, text?: string}>}  $data
     */
    public function handoff(array $data): Ticket
    {
        $history = [];

        foreach (array_slice((array) ($data['history'] ?? []), -30) as $item) {
            $text = trim((string) (is_array($item) ? ($item['text'] ?? '') : ''));
            $role = (string) (is_array($item) ? ($item['role'] ?? 'user') : 'user');

            if ($text === '') {
                continue;
            }

            $history[] = [
                'author' => $role === 'bot' ? Ticket::AUTHOR_BOT : Ticket::AUTHOR_CUSTOMER,
                'body' => mb_substr($text, 0, 2000),
            ];
        }

        $message = trim((string) ($data['message'] ?? ''));

        if ($message !== '') {
            $history[] = ['author' => Ticket::AUTHOR_CUSTOMER, 'body' => mb_substr($message, 0, 2000)];
        }

        $first = '';

        foreach ($history as $item) {
            if ($item['author'] === Ticket::AUTHOR_CUSTOMER) {
                $first = $item['body'];
                break;
            }
        }

        $email = mb_strtolower(trim((string) $data['email']));
        $name = trim((string) ($data['name'] ?? ''));
        $customer = Customer::query()->firstOrCreate(['email' => $email], ['name' => $name]);

        if ($name !== '' && trim((string) $customer->name) === '') {
            $customer->forceFill(['name' => $name])->save();
        }

        return $this->create([
            'customer_id' => (int) $customer->id,
            'email' => $email,
            'name' => $name,
            'subject' => mb_substr($first !== '' ? $first : 'Вопрос из Консультанта', 0, 120),
            'body' => '',
            'history' => $history,
            'source' => 'chat',
        ]);
    }

    /**
     * customerReply — сообщение покупателя в существующем обращении.
     *
     * @param  array<int, array<string, mixed>>  $attachments
     */
    public function customerReply(Ticket $ticket, string $body, string $name = '', array $attachments = []): TicketMessage
    {
        $message = $this->addMessage($ticket, Ticket::AUTHOR_CUSTOMER, $body, $name, $attachments);

        $ticket->status = Ticket::STATUS_OPEN;
        $ticket->closed_at = null;
        $ticket->save();

        $this->notifyAdmin($ticket);
        Telegram::notifyCustomerMessage($ticket, $message);

        return $message;
    }

    /**
     * supportReply — ответ поддержки. Первый ответ закрывает SLA.
     */
    public function supportReply(Ticket $ticket, string $body, string $authorName = 'Поддержка'): TicketMessage
    {
        $message = $this->addMessage($ticket, Ticket::AUTHOR_SUPPORT, $body, $authorName);

        if (!$ticket->first_response_at) {
            $ticket->first_response_at = now();
        }

        $ticket->status = Ticket::STATUS_WAITING;
        $ticket->closed_at = null;
        $ticket->save();

        $this->notifyCustomer($ticket, $message);

        return $message;
    }

    /**
     * close — обращение закрыто, в истории остаётся служебная запись.
     */
    public function close(Ticket $ticket, string $authorName = 'Поддержка', string $note = 'Обращение закрыто.'): void
    {
        if ($ticket->is_closed) {
            return;
        }

        $ticket->status = Ticket::STATUS_CLOSED;
        $ticket->closed_at = now();
        $ticket->save();

        $this->addMessage($ticket, Ticket::AUTHOR_SYSTEM, $note, $authorName);
    }

    /**
     * autoclose — закрывает обращения, где покупатель молчит после ответа.
     *
     * Считаются только обращения в статусе «Ждём ответа»: если последнее
     * сообщение написал покупатель, обращение в работе и закрывать его нельзя.
     */
    public function autoclose(int $days): int
    {
        if ($days <= 0) {
            return 0;
        }

        $closed = 0;
        $threshold = now()->subDays($days);

        Ticket::query()
            ->where('status', Ticket::STATUS_WAITING)
            ->where('updated_at', '<=', $threshold)
            ->with('messages')
            ->chunkById(100, function ($tickets) use (&$closed, $days) {
                foreach ($tickets as $ticket) {
                    $last = $ticket->messages->last();

                    if ($last && (string) $last->author === Ticket::AUTHOR_CUSTOMER) {
                        continue;
                    }

                    $this->close(
                        $ticket,
                        'Автозакрытие',
                        'Обращение закрыто автоматически: нет ответа покупателя ' . $days . ' дн.'
                    );

                    $closed++;
                }
            });

        return $closed;
    }

    /**
     * autocloseCount — сколько обращений попадёт под автозакрытие.
     */
    public function autocloseCount(int $days): int
    {
        if ($days <= 0) {
            return 0;
        }

        $count = 0;

        Ticket::query()
            ->where('status', Ticket::STATUS_WAITING)
            ->where('updated_at', '<=', now()->subDays($days))
            ->with('messages')
            ->chunkById(100, function ($tickets) use (&$count) {
                foreach ($tickets as $ticket) {
                    $last = $ticket->messages->last();

                    if ($last && (string) $last->author === Ticket::AUTHOR_CUSTOMER) {
                        continue;
                    }

                    $count++;
                }
            });

        return $count;
    }

    /**
     * quickReplies — шаблоны ответов оператора из настроек.
     *
     * Формат строки — «Название | Текст». Пустая настройка означает, что
     * шаблонов нет: кнопки в админке не показываются.
     *
     * @return array<int, array{title: string, body: string}>
     */
    public static function quickReplies(): array
    {
        $raw = (string) Settings::get('ticket_quick_replies', '');
        $items = [];

        foreach (preg_split('/\r\n|\r|\n/', $raw) ?: [] as $line) {
            $line = trim($line);

            if ($line === '' || !str_contains($line, '|')) {
                continue;
            }

            [$title, $body] = array_map('trim', explode('|', $line, 2));

            if ($title === '' || $body === '') {
                continue;
            }

            $items[] = [
                'title' => mb_substr($title, 0, 60),
                'body' => mb_substr($body, 0, 2000),
            ];
        }

        return array_slice($items, 0, 12);
    }

    /**
     * create — общая запись обращения и первого сообщения.
     *
     * @param  array<string, mixed>  $data
     */
    protected function create(array $data): Ticket
    {
        $email = mb_strtolower(trim((string) ($data['email'] ?? '')));
        $name = trim((string) ($data['name'] ?? ''));

        $ticket = new Ticket();
        $ticket->fill([
            'customer_id' => !empty($data['customer_id']) ? (int) $data['customer_id'] : null,
            'email' => $email,
            'order_id' => !empty($data['order_id']) ? (int) $data['order_id'] : null,
            'subject' => mb_substr(trim((string) ($data['subject'] ?? '')), 0, 200),
            'status' => Ticket::STATUS_NEW,
            'priority' => Ticket::PRIORITY_NORMAL,
            'source' => (string) ($data['source'] ?? 'account'),
        ]);
        $ticket->save();

        $history = (array) ($data['history'] ?? []);

        if ($history === []) {
            $this->addMessage(
                $ticket,
                Ticket::AUTHOR_CUSTOMER,
                (string) ($data['body'] ?? ''),
                $name,
                (array) ($data['attachments'] ?? [])
            );
        }
        else {
            foreach ($history as $item) {
                $this->addMessage(
                    $ticket,
                    (string) ($item['author'] ?? Ticket::AUTHOR_CUSTOMER),
                    (string) ($item['body'] ?? ''),
                    (string) ($item['author'] ?? '') === Ticket::AUTHOR_BOT ? 'Консультант' : $name
                );
            }
        }

        $this->notifyAdmin($ticket);
        Telegram::notifyNew($ticket);

        return $ticket->fresh();
    }

    /**
     * @param  array<int, array<string, mixed>>  $attachments
     */
    protected function addMessage(Ticket $ticket, string $author, string $body, string $name = '', array $attachments = []): TicketMessage
    {
        return $ticket->messages()->create([
            'author' => $author,
            'author_name' => mb_substr($name, 0, 160),
            'body' => trim($body),
            'attachments' => $attachments !== [] ? array_values($attachments) : null,
        ]);
    }

    protected function notifyAdmin(Ticket $ticket): void
    {
        $admin = (string) (Settings::get('order_notify_email') ?: Settings::get('email'));

        if ($admin === '') {
            Log::warning('Elv Kit: некуда отправить письмо о обращении', ['ticket' => $ticket->number]);

            return;
        }

        try {
            Mail::to($admin)->queue(new TicketAdmin($ticket));
        }
        catch (\Throwable $e) {
            Log::warning('Elv Kit: письмо о обращении не отправлено', [
                'ticket' => $ticket->number,
                'error' => $e->getMessage(),
            ]);
        }
    }

    protected function notifyCustomer(Ticket $ticket, TicketMessage $message): void
    {
        if (trim((string) $ticket->email) === '') {
            return;
        }

        try {
            Mail::to($ticket->email)->queue(new TicketCustomer($ticket, $message));
        }
        catch (\Throwable $e) {
            Log::warning('Elv Kit: письмо с ответом поддержки не отправлено', [
                'ticket' => $ticket->number,
                'error' => $e->getMessage(),
            ]);
        }
    }
}

<?php namespace Elv\Kit\Classes;

use Elv\Kit\Models\Settings;
use Elv\Kit\Models\Ticket;
use Illuminate\Http\UploadedFile;
use Illuminate\Support\Str;

/**
 * TicketFiles — вложения обращений: белый список, хранение и выдача.
 *
 * Файлы лежат в `storage/app/elv-kit/tickets/{id}/`, вне `public/`: прямой
 * ссылки у них нет, скачивание идёт через маршрут с проверкой прав. Имя на
 * диске случайное, исходное имя хранится только в метаданных, поэтому путь
 * нельзя подделать через параметры запроса.
 */
class TicketFiles
{
    public const MAX_FILES = 3;
    public const MAX_KB = 5120;

    /** @var string[] */
    public const ALLOWED = ['png', 'jpg', 'jpeg', 'gif', 'webp', 'pdf', 'txt', 'log', 'csv', 'zip'];

    /** @var string[] */
    public const ALLOWED_MIME = [
        'image/png', 'image/jpeg', 'image/webp', 'image/gif',
        'application/pdf', 'text/plain', 'text/csv', 'application/csv',
        'application/zip', 'application/x-zip-compressed',
        'application/octet-stream',
    ];

    public static function enabled(): bool
    {
        $value = Settings::get('ticket_attachments_enabled');

        // Пустое значение — это ещё не сохранённая настройка, а не «выключено»:
        // поле включено по умолчанию, поэтому пусто и null читаем как true.
        return $value === '' || $value === null ? true : (bool) $value;
    }

    /**
     * rules — правила валидации загрузки для формы обращения.
     *
     * @return array<string, string>
     */
    public static function rules(): array
    {
        $mimes = implode(',', self::ALLOWED);

        return [
            'attachments' => 'array|max:' . self::MAX_FILES,
            'attachments.*' => 'file|max:' . self::MAX_KB . '|mimes:' . $mimes,
        ];
    }

    /**
     * messages — человеческие тексты ошибок загрузки.
     *
     * @return array<string, string>
     */
    public static function messages(): array
    {
        return [
            'attachments.max' => 'К обращению можно приложить не больше ' . self::MAX_FILES . ' файлов.',
            'attachments.*.file' => 'Файл не загрузился. Попробуйте ещё раз.',
            'attachments.*.max' => 'Файл больше ' . (self::MAX_KB / 1024) . ' МБ.',
            'attachments.*.mimes' => 'Такой тип файла не принимается: ' . implode(', ', self::ALLOWED) . '.',
        ];
    }

    /**
     * store — сохраняет загруженные файлы и возвращает метаданные для БД.
     *
     * @param  array<int, UploadedFile>  $files
     * @return array<int, array{name: string, size: int, ext: string, file: string}>
     */
    public function store(Ticket $ticket, array $files): array
    {
        if (!self::enabled() || $files === []) {
            return [];
        }

        $dir = $this->directory((int) $ticket->id);

        if (!is_dir($dir) && !@mkdir($dir, 0755, true) && !is_dir($dir)) {
            return [];
        }

        $saved = [];

        foreach (array_slice($files, 0, self::MAX_FILES) as $file) {
            if (!$file instanceof UploadedFile || !$file->isValid()) {
                continue;
            }

            $ext = mb_strtolower((string) $file->getClientOriginalExtension());
            $mime = mb_strtolower((string) $file->getMimeType());

            if (
                !in_array($ext, self::ALLOWED, true)
                || !in_array($mime, self::ALLOWED_MIME, true)
                || $file->getSize() > self::MAX_KB * 1024
            ) {
                continue;
            }

            $stored = Str::random(32) . '.' . $ext;
            $name = $this->cleanName((string) $file->getClientOriginalName(), $ext);
            $size = (int) $file->getSize();

            if (!$file->move($dir, $stored)) {
                continue;
            }

            $saved[] = [
                'name' => $name,
                'size' => $size,
                'ext' => $ext,
                'file' => $stored,
            ];
        }

        return $saved;
    }

    /**
     * path — полный путь к файлу по имени из метаданных.
     */
    public function path(int $ticketId, string $stored): ?string
    {
        $stored = basename($stored);

        if (!preg_match('/^[A-Za-z0-9]{32}\.[a-z0-9]{2,5}$/', $stored)) {
            return null;
        }

        $path = $this->directory($ticketId) . DIRECTORY_SEPARATOR . $stored;

        return is_file($path) ? $path : null;
    }

    /**
     * find — метаданные вложения по имени файла на диске.
     *
     * @param  array<int, array<string, mixed>>  $attachments
     * @return array<string, mixed>|null
     */
    public static function find(array $attachments, string $stored): ?array
    {
        foreach ($attachments as $item) {
            if (is_array($item) && (string) ($item['file'] ?? '') === $stored) {
                return $item;
            }
        }

        return null;
    }

    public static function sizeLabel(int $bytes): string
    {
        if ($bytes >= 1024 * 1024) {
            return round($bytes / 1024 / 1024, 1) . ' МБ';
        }

        if ($bytes >= 1024) {
            return round($bytes / 1024) . ' КБ';
        }

        return $bytes . ' Б';
    }

    protected function directory(int $ticketId): string
    {
        return storage_path('app/elv-kit/tickets/' . $ticketId);
    }

    protected function cleanName(string $name, string $ext): string
    {
        $name = basename($name);
        $name = preg_replace('/[^\p{L}\p{N}._ -]+/u', '_', $name) ?: 'file.' . $ext;
        $name = trim($name);

        if ($name === '' || $name === '.' . $ext) {
            $name = 'file.' . $ext;
        }

        return mb_substr($name, 0, 120);
    }
}

    {
        return static::token() !== '' && static::chatId() !== '';

    {

        return $expected !== '' && hash_equals($expected, trim($secret));

    {

<?php namespace Elv\Kit\Classes\Http;

use Elv\Kit\Classes\AccountService;
use Elv\Kit\Classes\AntiSpam;
use Elv\Kit\Classes\TicketFiles;
use Elv\Kit\Classes\TicketService;
use Elv\Kit\Models\Ticket;
use Elv\Kit\Models\TicketMessage;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Validator;

/**
 * TicketController — обращения покупателя: создать, ответить, закрыть.
 *
 * Доступ только из кабинета: покупатель опознаётся по сессии Magic Link, а
 * обращение проверяется по `customer_id`/e-mail. Чужое обращение отвечает 404,
 * как несуществующее.
 */
class TicketController
{
    public function __construct(
        protected AccountService $account = new AccountService(),
        protected TicketService $tickets = new TicketService()
    ) {
    }

    /**
     * store создаёт обращение с первым сообщением.
     *
     * @return \Illuminate\Http\JsonResponse|\Illuminate\Http\RedirectResponse
     */
    public function store(Request $request)
    {
        $customer = $this->account->user();

        if (!$customer) {
            return $this->denied($request);
        }

        if (AntiSpam::isBot($request)) {
            return $this->created($request, null);
        }

        $validator = Validator::make($request->all(), array_merge([
            'subject' => 'required|string|min:5|max:200',
            'body' => 'required|string|min:10|max:5000',
            'order_id' => 'nullable|integer|min:1',
            'consent' => 'accepted',
        ], TicketFiles::enabled() ? TicketFiles::rules() : []), array_merge([
            'subject.required' => 'Укажите тему обращения.',
            'subject.min' => 'Тема слишком короткая.',
            'body.required' => 'Опишите вопрос.',
            'body.min' => 'Опишите вопрос подробнее — минимум 10 символов.',
            'consent.accepted' => 'Нужно согласие на обработку персональных данных.',
        ], TicketFiles::messages()));

        if ($validator->fails()) {
            return $this->invalid($request, $validator->errors()->toArray());
        }

        $orderId = (int) $request->input('order_id', 0);
        $order = null;

        if ($orderId > 0) {
            $order = $this->account->orders($customer)->firstWhere('id', $orderId);

            if (!$order) {
                return $this->invalid($request, ['order_id' => ['Заказ не найден среди ваших заказов.']]);
            }
        }

        $ticket = $this->tickets->open($customer, [
            'subject' => AntiSpam::clean((string) $request->input('subject'), 200),
            'body' => AntiSpam::clean((string) $request->input('body'), 5000),
            'order_id' => $order ? (int) $order->id : null,
        ]);

        $this->attachFiles($ticket, $request, $ticket->messages()->first());

        return $this->created($request, $ticket);
    }

    /**
     * reply добавляет сообщение покупателя в обращение.
     *
     * @return \Illuminate\Http\JsonResponse|\Illuminate\Http\RedirectResponse
     */
    public function reply(Request $request, int $id)
    {
        $customer = $this->account->user();

        if (!$customer) {
            return $this->denied($request);
        }

        $ticket = $this->account->ticket($customer, $id);

        if (!$ticket) {
            abort(404);
        }

        if (AntiSpam::isBot($request)) {
            return $this->replied($request, $id);
        }

        $validator = Validator::make($request->all(), array_merge([
            'body' => 'required|string|min:2|max:5000',
        ], TicketFiles::enabled() ? TicketFiles::rules() : []), array_merge([
            'body.required' => 'Напишите сообщение.',
            'body.min' => 'Сообщение слишком короткое.',
        ], TicketFiles::messages()));

        if ($validator->fails()) {
            return $this->invalid($request, $validator->errors()->toArray(), $id);
        }

        $files = (new TicketFiles())->store($ticket, (array) $request->file('attachments', []));

        $this->tickets->customerReply(
            $ticket,
            AntiSpam::clean((string) $request->input('body'), 5000),
            $customer->display_name,
            $files
        );

        return $this->replied($request, $id);
    }

    /**
     * close закрывает обращение по просьбе покупателя.
     *
     * @return \Illuminate\Http\RedirectResponse
     */
    public function close(Request $request, int $id)
    {
        $customer = $this->account->user();

        if (!$customer) {
            return $this->denied($request);
        }

        $ticket = $this->account->ticket($customer, $id);

        if (!$ticket) {
            abort(404);
        }

        $this->tickets->close($ticket, $customer->display_name);

        return redirect(url('/tickets') . '/?id=' . (int) $ticket->id . '&closed=1#thread')
            ->with('elv_success', 'Обращение закрыто. Если понадобится, напишите снова — оно откроется.');
    }

    /**
     * thread отдаёт новые сообщения для лёгкого polling открытой вкладки.
     *
     * Возвращаются только сообщения после `since`, поэтому запрос дешёвый и не
     * перерисовывает уже прочитанное. HTML не собирается на сервере: браузер
     * строит узлы сам, текст вставляется как текст.
     */
    public function thread(Request $request, int $id)
    {
        $customer = $this->account->user();

        if (!$customer) {
            return response()->json(['ok' => false, 'code' => 'AUTH'], 401);
        }

        $ticket = $this->account->ticket($customer, $id);

        if (!$ticket) {
            abort(404);
        }

        $since = max(0, (int) $request->query('since', 0));
        $messages = [];

        foreach ($ticket->messages as $message) {
            if ((int) $message->id <= $since) {
                continue;
            }

            $messages[] = [
                'id' => (int) $message->id,
                'author' => (string) $message->author,
                'author_label' => (string) $message->author_label,
                'author_name' => (string) ($message->author_name ?? ''),
                'body' => (string) $message->body,
                'date' => $message->created_at ? $message->created_at->format('d.m.Y H:i') : '',
                'is_support' => (bool) $message->is_support,
                'files' => $message->files,
            ];
        }

        return response()->json([
            'ok' => true,
            'messages' => $messages,
            'status' => (string) $ticket->status,
            'status_label' => (string) $ticket->status_label,
            'closed' => (bool) $ticket->is_closed,
            'sla_label' => (string) $ticket->sla_label,
            'sla_overdue' => (bool) $ticket->sla_overdue,
        ], 200, [], JSON_UNESCAPED_UNICODE);
    }

    /**
     * file отдаёт вложение обращения с проверкой прав.
     *
     * Покупатель видит только свои обращения; сотрудник поддержки — любое,
     * если у него есть право `elv.kit.tickets`. Файлы лежат вне `public/`,
     * поэтому другого пути к ним нет.
     *
     * @return \Symfony\Component\HttpFoundation\BinaryFileResponse
     */
    public function file(Request $request, int $id, string $name)
    {
        $ticket = Ticket::query()->with('messages')->find($id);
        $customer = $this->account->user();
        $allowed = $ticket && (
            ($customer && $this->account->ticket($customer, $id) !== null)
            || (($user = \BackendAuth::getUser()) && $user->hasPermission('elv.kit.tickets'))
        );

        if (!$ticket || !$allowed) {
            abort(404);
        }

        $meta = null;

        foreach ($ticket->messages as $message) {
            $meta = TicketFiles::find((array) $message->attachments, $name);

            if ($meta) {
                break;
            }
        }

        $path = $meta ? (new TicketFiles())->path((int) $ticket->id, $name) : null;

        if (!$meta || !$path) {
            abort(404);
        }

        return response()->download($path, (string) ($meta['name'] ?? $name), [
            'Content-Type' => 'application/octet-stream',
        ]);
    }

    /**
     * attachFiles — вложения первого сообщения: файл сохраняется уже после
     * создания обращения, потому что папка называется по его номеру.
     */
    protected function attachFiles(Ticket $ticket, Request $request, ?TicketMessage $message): void
    {
        if (!$message || !TicketFiles::enabled()) {
            return;
        }

        $files = (new TicketFiles())->store($ticket, (array) $request->file('attachments', []));

        if ($files === []) {
            return;
        }

        $message->attachments = $files;
        $message->save();
    }

    protected function created(Request $request, $ticket)
    {
        if ($request->expectsJson()) {
            return response()->json([
                'ok' => true,
                'id' => $ticket ? (int) $ticket->id : null,
                'number' => $ticket ? (string) $ticket->number : '',
                'url' => $ticket ? (string) $ticket->url : '',
            ], 201);
        }

        $url = $ticket
            ? url('/tickets') . '/?id=' . (int) $ticket->id . '&created=1#thread'
            : url('/tickets') . '/?created=1';

        return redirect($url)->with('elv_success', 'Обращение создано. Ответ придёт на e-mail и появится здесь.');
    }

    protected function replied(Request $request, int $id)
    {
        if ($request->expectsJson()) {
            return response()->json(['ok' => true], 201);
        }

        return redirect(url('/tickets') . '/?id=' . $id . '&sent=1#thread')
            ->with('elv_success', 'Сообщение отправлено.');
    }

    /**
     * @param  array<string, mixed>  $errors
     * @return \Illuminate\Http\JsonResponse|\Illuminate\Http\RedirectResponse
     */
    protected function invalid(Request $request, array $errors, ?int $id = null)
    {
        if ($request->expectsJson()) {
            return response()->json([
                'ok' => false,
                'code' => 'VALIDATION',
                'errors' => $errors,
            ], 422);
        }

        $first = '';

        foreach ($errors as $messages) {
            $first = (string) (is_array($messages) ? reset($messages) : $messages);
            break;
        }

        $url = url('/tickets') . '/' . ($id ? '?id=' . $id . '#thread' : '#new-ticket');

        return redirect($url)
            ->withInput()
            ->withErrors($errors)
            ->with('elv_errors', $errors)
            ->with('elv_error', $first);
    }

    protected function denied(Request $request)
    {
        if ($request->expectsJson()) {
            return response()->json([
                'ok' => false,
                'code' => 'AUTH',
                'message' => 'Войдите в кабинет, чтобы написать в поддержку.',
            ], 401);
        }

        return redirect(url('/account') . '/')
            ->with('elv_error', 'Войдите в кабинет, чтобы написать в поддержку.');
    }
}

<?php namespace Elv\Kit\Classes\Http;

use Elv\Kit\Classes\AccountService;
use Elv\Kit\Classes\AntiSpam;
use Elv\Kit\Classes\Chat;
use Elv\Kit\Classes\TicketService;
use Illuminate\Http\Request;
use Illuminate\Routing\Controller;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Support\Facades\Validator;

/**
 * ChatController — ответы Консультанта по базе знаний сайта.
 *
 * Внешних API нет: ответ собирается из каталога и настроек, поэтому работает
 * и на демо-контуре без ключей. Лимит запросов задан в routes.php.
 *
 * Переписка хранится в сессии: кнопка «Передать оператору» отправляет её в
 * поддержку вместе с контактами, поэтому покупателю не нужно пересказывать
 * вопрос заново. История обрезается до последних сообщений и живёт до отправки
 * обращения.
 */
class ChatController extends Controller
{
    public const HISTORY_KEY = 'elv_chat_history';
    public const HISTORY_LIMIT = 30;

    public function store(Request $request)
    {
        if (AntiSpam::isBot($request)) {
            return AntiSpam::json([
                'ok' => true,
                'answer' => 'Здравствуйте! Чем помочь по Elv Kit?',
                'chips' => ['Сколько стоит?', 'Как установить?'],
                'link' => null,
                'open' => null,
                'sources' => [],
                'rejected' => false,
            ]);
        }

        $validator = Validator::make($request->all(), [
            'message' => 'required|string|min:2|max:' . Chat::MAX_LENGTH,
        ], [
            'message.required' => 'Напишите вопрос.',
            'message.min' => 'Напишите вопрос подробнее — от двух символов.',
            'message.max' => 'Слишком длинный вопрос: до ' . Chat::MAX_LENGTH . ' символов.',
        ]);

        if ($validator->fails()) {
            return AntiSpam::json([
                'ok' => false,
                'code' => 'VALIDATION',
                'errors' => $validator->errors()->toArray(),
            ], 422);
        }

        $key = 'elv-chat:' . $request->session()->getId();

        if (RateLimiter::tooManyAttempts($key, 15)) {
            return AntiSpam::json([
                'ok' => false,
                'code' => 'RATE_LIMIT',
                'message' => 'Слишком много сообщений подряд. Подождите минуту.',
            ], 429)->header('Retry-After', (string) RateLimiter::availableIn($key));
        }

        RateLimiter::hit($key, 60);

        $message = (string) $request->input('message');
        $result = Chat::answer($message);

        // Отклонённые реплики (бессмыслица, грубость) в историю не попадают:
        // оператору нужен диалог по делу, а не мусор.
        if (empty($result['rejected']) && empty($result['hide_user'])) {
            $this->remember('user', $message);
            $this->remember('bot', (string) ($result['answer'] ?? ''));
        }

        return AntiSpam::json(['ok' => true] + $result);
    }

    /**
     * handoff создаёт обращение из переписки Консультанта.
     *
     * Ответы бота сохраняются отдельным автором: поддержка видит, что уже
     * сказал консультант. После отправки история очищается, чтобы следующее
     * обращение не приносило старые сообщения.
     */
    public function handoff(Request $request)
    {
        if (AntiSpam::isBot($request)) {
            return AntiSpam::json(['ok' => true, 'number' => '', 'url' => null]);
        }

        $key = 'elv-handoff:' . $request->session()->getId();

        if (RateLimiter::tooManyAttempts($key, 5)) {
            return AntiSpam::json([
                'ok' => false,
                'code' => 'RATE_LIMIT',
                'message' => 'Слишком много попыток. Подождите несколько минут.',
            ], 429)->header('Retry-After', (string) RateLimiter::availableIn($key));
        }

        RateLimiter::hit($key, 600);

        $validator = Validator::make($request->all(), [
            'name' => 'required|string|max:160',
            'email' => 'required|email|max:191',
            'message' => 'nullable|string|max:2000',
            'consent' => 'accepted',
        ], [
            'name.required' => 'Укажите имя.',
            'email.required' => 'Укажите e-mail для ответа.',
            'email.email' => 'Проверьте адрес электронной почты.',
            'consent.accepted' => 'Нужно согласие на обработку персональных данных.',
        ]);

        if ($validator->fails()) {
            return AntiSpam::json([
                'ok' => false,
                'code' => 'VALIDATION',
                'errors' => $validator->errors()->toArray(),
            ], 422);
        }

        $ticket = (new TicketService())->handoff([
            'email' => (string) $request->input('email'),
            'name' => AntiSpam::clean((string) $request->input('name'), 160),
            'message' => AntiSpam::clean((string) $request->input('message', ''), 2000),
            'history' => (array) $request->session()->get(self::HISTORY_KEY, []),
        ]);

        $request->session()->forget(self::HISTORY_KEY);

        $customer = (new AccountService())->user();

        return AntiSpam::json([
            'ok' => true,
            'number' => (string) $ticket->number,
            'url' => $customer ? (string) $ticket->url : null,
        ]);
    }

    protected function remember(string $role, string $text): void
    {
        $text = trim($text);

        if ($text === '') {
            return;
        }

        $history = (array) session()->get(self::HISTORY_KEY, []);
        $history[] = ['role' => $role === 'bot' ? 'bot' : 'user', 'text' => mb_substr($text, 0, 2000)];

        session()->put(self::HISTORY_KEY, array_slice($history, -self::HISTORY_LIMIT));
    }
}

use Elv\Kit\Classes\Telegram;
use Elv\Kit\Classes\TicketService;
use Elv\Kit\Models\Ticket;
use Illuminate\Http\Request;

/**
 * TelegramController — вебхук ChatOps.
 *
 * Команды принимаются только из настроенного чата и только с секретом в
 * адресе. Ответ покупателю уходит письмом, а подтверждение — в Telegram.

 */

{

    }

    {

            abort(404);
        }

        $update = (array) $request->json()->all();
        $message = (array) ($update['message'] ?? $update['edited_message'] ?? []);
        $chatId = (string) ($message['chat']['id'] ?? '');

        if (!Telegram::chatMatches($chatId)) {
            return response()->json(['ok' => true]);
        }

        $text = trim((string) ($message['text'] ?? ''));

        if ($text === '') {
            return response()->json(['ok' => true]);
        }

        $author = trim((string) ($message['from']['first_name'] ?? '')) ?: 'Поддержка';

        if (preg_match('~^/(start|help)\b~u', $text)) {
            Telegram::send(Telegram::helpText());

            return response()->json(['ok' => true]);
        }

        if (preg_match('~^/reply\s+(TK-\d{6})\s+([\s\S]+)$~u', $text, $match)) {
            $ticket = Ticket::query()->where('number', $match[1])->first();

            if (!$ticket) {
                Telegram::send('Обращение ' . $match[1] . ' не найдено.');

                return response()->json(['ok' => true]);
            }

            $this->tickets->supportReply($ticket, trim($match[2]), $author);
            Telegram::send('Ответ отправлен в ' . $ticket->number . '. Статус: ждём ответа покупателя.');

            return response()->json(['ok' => true]);
        }

        if (preg_match('~^/close\s+(TK-\d{6})$~u', $text, $match)) {
            $ticket = Ticket::query()->where('number', $match[1])->first();

            if (!$ticket) {
                Telegram::send('Обращение ' . $match[1] . ' не найдено.');

                return response()->json(['ok' => true]);
            }

            $this->tickets->close($ticket, $author);
            Telegram::send('Обращение ' . $ticket->number . ' закрыто.');

            return response()->json(['ok' => true]);
        }

        if (preg_match('~^/status\s+(TK-\d{6})$~u', $text, $match)) {
            $ticket = Ticket::query()->where('number', $match[1])->first();

            if (!$ticket) {
                Telegram::send('Обращение ' . $match[1] . ' не найдено.');

                return response()->json(['ok' => true]);
            }

            Telegram::send(implode("\n", [
                $ticket->number . ' — ' . $ticket->status_label,
                'Тема: ' . $ticket->subject,
                'Покупатель: ' . $ticket->email,
                $ticket->sla_label,
            ]));

            return response()->json(['ok' => true]);
        }

        Telegram::send('Не понял команду. ' . "\n" . Telegram::helpText());

        return response()->json(['ok' => true]);
    }
}

<?php namespace Elv\Kit\Console;

use Elv\Kit\Classes\TicketService;
use Elv\Kit\Models\Settings;
use Illuminate\Console\Command;

/**
 * AutocloseCommand — закрывает обращения, где покупатель молчит после ответа.
 *
 * Срок берётся из настроек (`ticket_autoclose_days`, по умолчанию 5 дней,
 * `0` — не закрывать). Обращения в работе не трогаются: если последнее слово
 * осталось за покупателем, поддержка ещё должна ответить.
 */
class AutocloseCommand extends Command
{
    protected $signature = 'elv:kit:autoclose
        {--days= : срок молчания покупателя в днях, перекрывает настройку}
        {--dry : только показать, сколько обращений будет закрыто}';

    protected $description = 'Закрывает обращения без ответа покупателя.';

    public function handle()
    {
        $days = $this->option('days');
        $days = $days === null ? (int) Settings::get('ticket_autoclose_days', 5) : (int) $days;

        if ($days <= 0) {
            $this->line('Автозакрытие выключено в настройках.');

            return 0;
        }

        if ($this->option('dry')) {
            $count = (new TicketService())->autocloseCount($days);
            $this->info('Будет закрыто обращений: ' . $count . ' (срок ' . $days . ' дн.).');

            return 0;
        }

        $closed = (new TicketService())->autoclose($days);
        $this->info('Закрыто обращений: ' . $closed . ' (срок ' . $days . ' дн.).');

        return 0;
    }
}

<?php namespace Elv\Kit\Models;

use Elv\Kit\Classes\Support;
use Model;

/**
 * Ticket — обращение покупателя в поддержку.
 *
 * Номер вида `TK-000123` присваивается после вставки: он нужен в письмах,
 * Telegram и админке, но не должен зависеть от ввода покупателя. Срок первого
 * ответа считается по рабочим часам (`Support::dueAt()`).
 */
class Ticket extends Model
{
    public const STATUS_NEW = 'new';
    public const STATUS_OPEN = 'open';
    public const STATUS_WAITING = 'waiting';
    public const STATUS_CLOSED = 'closed';

    public const PRIORITY_LOW = 'low';
    public const PRIORITY_NORMAL = 'normal';
    public const PRIORITY_HIGH = 'high';

    public const AUTHOR_CUSTOMER = 'customer';
    public const AUTHOR_SUPPORT = 'support';
    public const AUTHOR_SYSTEM = 'system';
    public const AUTHOR_BOT = 'bot';

    public $table = 'elv_kit_tickets';

    public $fillable = [
        'number', 'customer_id', 'email', 'order_id', 'subject',
        'status', 'priority', 'source', 'sla_due_at', 'first_response_at', 'closed_at',
    ];

    public $rules = [
        'email' => 'required|email|max:191',
        'subject' => 'required|max:200',
        'status' => 'required|in:new,open,waiting,closed',
        'priority' => 'required|in:low,normal,high',
    ];

    protected $casts = [
        'sla_due_at' => 'datetime',
        'first_response_at' => 'datetime',
        'closed_at' => 'datetime',
    ];

    public $belongsTo = [
        'customer' => Customer::class,
        'order' => Order::class,
    ];

    public $hasMany = [
        'messages' => [TicketMessage::class, 'key' => 'ticket_id', 'order' => 'id asc'],
    ];

    protected static function booted()
    {
        static::creating(function (self $ticket) {
            if (!$ticket->sla_due_at) {
                $ticket->sla_due_at = Support::dueAt(now());
            }
        });

        static::created(function (self $ticket) {
            if (trim((string) $ticket->number) === '') {
                $ticket->number = 'TK-' . str_pad((string) $ticket->id, 6, '0', STR_PAD_LEFT);
                $ticket->saveQuietly();
            }
        });
    }

    /**
     * forCustomer — обращения покупателя по id или e-mail.
     */
    public function scopeForCustomer($query, Customer $customer)
    {
        return $query->where(function ($inner) use ($customer) {
            $inner->where('customer_id', (int) $customer->id)
                ->orWhere('email', $customer->email);
        });
    }

    public function scopeUnanswered($query)
    {
        return $query->whereIn('status', [self::STATUS_NEW, self::STATUS_OPEN]);
    }

    public function getStatusLabelAttribute(): string
    {
        return match ((string) $this->status) {
            self::STATUS_OPEN => 'В работе',
            self::STATUS_WAITING => 'Ждём ответа',
            self::STATUS_CLOSED => 'Закрыто',
            default => 'Новое',
        };
    }

    public function getPriorityLabelAttribute(): string
    {
        return match ((string) $this->priority) {
            self::PRIORITY_HIGH => 'Высокий',
            self::PRIORITY_LOW => 'Низкий',
            default => 'Обычный',
        };
    }

    public function getSourceLabelAttribute(): string
    {
        return match ((string) $this->source) {
            'feedback' => 'Форма обратной связи',
            'chat' => 'Консультант',
            'telegram' => 'Telegram',
            default => 'Кабинет',
        };
    }

    /**
     * lastMessage — последнее сообщение переписки, если она загружена.
     */
    public function getLastMessageAttribute(): ?TicketMessage
    {
        return $this->messages->last();
    }

    public function getIsClosedAttribute(): bool
    {
        return (string) $this->status === self::STATUS_CLOSED;
    }

    public function getUrlAttribute(): string
    {
        // url() срезает слэш на конце, поэтому канонический адрес собирается
        // из двух частей: иначе каждая ссылка на обращение даёт 301.
        return url('/tickets') . '/?id=' . (int) $this->id;
    }

    public function getSlaLabelAttribute(): string
    {
        return Support::describe($this);
    }

    public function getSlaOverdueAttribute(): bool
    {
        return Support::isOverdue($this);
    }

    public function getLastMessageAtAttribute()
    {
        $message = $this->messages->last();

        return $message ? $message->created_at : $this->created_at;
    }

}

<?php namespace Elv\Kit\Models;

use Elv\Kit\Classes\TicketFiles;
use Model;

/**
 * TicketMessage — сообщение в обращении.
 *
 * Автор хранится строкой, а не ссылкой на пользователя: покупатель входит по
 * ссылке без учётной записи, а сотрудник поддержки может смениться. Имя
 * сохраняется на момент отправки, чтобы история не переписывалась.
 *
 * Вложения лежат списком в `attachments`; файл на диске открывается только
 * через маршрут с проверкой прав, поэтому в шаблоне выводится не путь, а
 * ссылка и исходное имя.
 */
class TicketMessage extends Model
{
    public $table = 'elv_kit_ticket_messages';

    public $fillable = [
        'ticket_id', 'author', 'author_name', 'body', 'attachments',
    ];

    public $rules = [
        'ticket_id' => 'required|integer',
        'author' => 'required|in:customer,support,system,bot',
        'body' => 'required|max:5000',
    ];

    protected $casts = [
        'attachments' => 'array',
    ];

    public $belongsTo = [
        'ticket' => Ticket::class,
    ];

    public function getAuthorLabelAttribute(): string
    {
        return match ((string) $this->author) {
            Ticket::AUTHOR_SUPPORT => 'Поддержка',
            Ticket::AUTHOR_SYSTEM => 'Система',
            Ticket::AUTHOR_BOT => 'Консультант',
            default => 'Покупатель',
        };
    }

    public function getIsSupportAttribute(): bool
    {
        return (string) $this->author === Ticket::AUTHOR_SUPPORT;
    }

    public function getIsBotAttribute(): bool
    {
        return (string) $this->author === Ticket::AUTHOR_BOT;
    }

    /**
     * files — вложения сообщения с готовой ссылкой на скачивание.
     *
     * @return array<int, array{name: string, size: int, size_label: string, url: string}>
     */
    public function getFilesAttribute(): array
    {
        $files = [];

        foreach ((array) $this->attachments as $item) {
            if (!is_array($item) || empty($item['file'])) {
                continue;
            }

            $files[] = [
                'name' => (string) ($item['name'] ?? $item['file']),
                'size' => (int) ($item['size'] ?? 0),
                'size_label' => TicketFiles::sizeLabel((int) ($item['size'] ?? 0)),
                'url' => url('/tickets') . '/' . (int) $this->ticket_id . '/file/' . rawurlencode((string) $item['file']),
            ];
        }

        return $files;
    }
}

/*
 * Поддержка: лёгкий polling открытой вкладки обращения.
 *
 * Раз в 12 секунд страница спрашивает только новые сообщения (`?since=`),
 * поэтому трафик небольшой и уже прочитанное не перерисовывается. Узлы
 * строятся через DOM, текст вставляется как текст: HTML из ответа сервера не
 * исполняется.
 */
(function () {
    'use strict';

    var thread = document.querySelector('[data-ticket-thread]');
    var url = thread ? thread.getAttribute('data-poll-url') : '';

    if (!thread || !url) {
        return;
    }

    var lastId = Number(thread.getAttribute('data-last-id')) || 0;
    var closed = false;
    var timer = null;

    var el = function (tag, className, text) {
        var node = document.createElement(tag);

        if (className) {
            node.className = className;
        }

        if (text !== undefined && text !== null) {
            node.textContent = String(text);
        }

        return node;
    };

    var addMessage = function (message) {
        var article = el('article', 'ticket-message' + (message.is_support ? ' ticket-message--support' : ''));
        var header = el('header');

        header.appendChild(el('b', '', message.author_label));

        if (message.author_name) {
            header.appendChild(el('span', 'tiny muted', ' · ' + message.author_name));
        }

        header.appendChild(el('span', 'tiny muted', ' · ' + message.date));
        article.appendChild(header);
        article.appendChild(el('div', 'ticket-message__body', message.body));

        if (message.files && message.files.length) {
            var list = el('ul', 'ticket-files');

            message.files.forEach(function (file) {
                var item = el('li');
                var link = el('a', '', file.name);

                link.href = file.url;
                link.setAttribute('download', '');
                item.appendChild(link);
                item.appendChild(el('span', 'tiny muted', ' ' + file.size_label));
                list.appendChild(item);
            });

            article.appendChild(list);
        }

        thread.appendChild(article);
        lastId = Math.max(lastId, Number(message.id) || 0);
    };

    var stop = function () {
        if (timer) {
            window.clearInterval(timer);
            timer = null;
        }
    };

    var applyClosed = function (data) {
        closed = true;
        stop();

        var badge = document.querySelector('[data-ticket-status]');
        if (badge && data.status_label) {
            badge.textContent = data.status_label;
            badge.className = 'badge badge--line';
        }

        var form = document.getElementById('reply');
        if (form) {
            form.style.display = 'none';
        }

        var note = document.querySelector('[data-ticket-closed]');

        if (!note && thread.parentNode) {
            note = el('div', 'notice mt-24');
            note.setAttribute('data-ticket-closed', '');
            note.appendChild(el('p', '', 'Обращение закрыто. Новое сообщение откроет его снова.'));
            thread.parentNode.appendChild(note);
        }
        else if (note) {
            note.hidden = false;
        }
    };

    var poll = function () {
        if (closed || document.hidden) {
            return;
        }

        fetch(url + '?since=' + lastId, {
            headers: {
                'Accept': 'application/json',
                'X-Requested-With': 'XMLHttpRequest'
            },
            credentials: 'same-origin'
        })
            .then(function (response) {
                if (!response.ok) {
                    return null;
                }

                return response.json().catch(function () {
                    return null;
                });
            })
            .then(function (data) {
                if (!data || !data.ok) {
                    return;
                }

                (data.messages || []).forEach(addMessage);

                if (data.closed) {
                    applyClosed(data);
                }
            })
            .catch(function () {
                return null;
            });
    };

    timer = window.setInterval(poll, 12000);

    document.addEventListener('visibilitychange', function () {
        if (!document.hidden) {
            poll();
        }
    });
})();
