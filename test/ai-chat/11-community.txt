<?php namespace Elv\Kit\Classes;

/**
 * Community — общие правила публичных вопросов и отзывов.
 *
 * Имя покупателя не публикуется целиком: на витрине остаётся имя и первая
 * буква фамилии («Алексей К.»). E-mail и IP хранятся только хэшем.
 */
class Community
{
    public static function displayName(string $name): string
    {
        $name = trim(preg_replace('/\s+/u', ' ', strip_tags($name)) ?? '');

        if ($name === '') {
            return 'Покупатель';
        }

        $parts = preg_split('/\s+/u', $name) ?: [];
        $first = (string) ($parts[0] ?? '');

        if ($first === '') {
            return 'Покупатель';
        }

        $last = count($parts) > 1 ? (string) end($parts) : '';
        $first = mb_substr($first, 0, 60);

        if ($last === '') {
            return $first;
        }

        return $first . ' ' . mb_strtoupper(mb_substr($last, 0, 1)) . '.';
    }

    public static function emailHash(string $email): string
    {
        return hash('sha256', mb_strtolower(trim($email)));
    }

    public static function ipHash(string $ip): string
    {
        return hash('sha256', trim($ip) . '|' . (string) config('app.key'));
    }

    public static function average(iterable $ratings): float
    {
        $sum = 0;
        $count = 0;

        foreach ($ratings as $rating) {
            $sum += (int) $rating;
            $count++;
        }

        return $count > 0 ? round($sum / $count, 1) : 0.0;
    }
}

<?php namespace Elv\Kit\Classes;

use Elv\Kit\Classes\Mail\QuestionAdmin;
use Elv\Kit\Models\Product;
use Elv\Kit\Models\Question;
use Elv\Kit\Models\Settings;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\Mail;

/**
 * Questions — публичные вопросы к продукту и их модерация.
 *
 * Вопрос сохраняется сразу, письмо администратору ставится после записи.
 * На витрину вопрос попадает только после ответа: `answer()` сам ставит
 * `published`, потому что ответ и есть модерация.
 */
class Questions
{
    public static function enabled(): bool
    {
        return filter_var(Settings::get('qa_enabled', true), FILTER_VALIDATE_BOOLEAN);
    }

    /**
     * published — вопросы продукта, которые можно показывать.
     *
     * @return \October\Rain\Database\Collection<int, Question>
     */
    public static function published(Product $product, int $limit = 0)
    {
        $query = Question::query()
            ->published()
            ->where('product_id', (int) $product->id);

        if ($limit > 0) {
            $query->limit($limit);
        }

        return $query->get();
    }

    public static function count(Product $product): int
    {
        return Question::query()
            ->where('product_id', (int) $product->id)
            ->where('status', Question::STATUS_PUBLISHED)
            ->whereNotNull('answer')
            ->where('answer', '<>', '')
            ->count();
    }

    /**
     * ask — новый вопрос. Статус `new`, на витрине его ещё нет.
     *
     * @param  array{name: string, email: string, question: string}  $data
     */
    public function ask(Product $product, array $data, string $ip = ''): Question
    {
        $question = new Question();
        $question->fill([
            'product_id' => (int) $product->id,
            'name' => (string) $data['name'],
            'email' => mb_strtolower((string) $data['email']),
            'question' => (string) $data['question'],
            'status' => Question::STATUS_NEW,
            'ip_hash' => Community::ipHash($ip),
        ]);
        $question->save();

        $this->notifyAdmin($question);

        return $question;
    }

    /**
     * answer — ответ поддержки. Публикует вопрос.
     */
    public function answer(Question $question, string $answer, string $author = 'Поддержка'): Question
    {
        $answer = trim($answer);

        if ($answer === '') {
            return $question;
        }

        $question->answer = $answer;
        $question->answered_by = mb_substr($author, 0, 160);
        $question->answered_at = now();
        $question->status = Question::STATUS_PUBLISHED;
        $question->published_at = now();
        $question->save();

        return $question->fresh();
    }

    public function reject(Question $question): Question
    {
        $question->status = Question::STATUS_REJECTED;
        $question->published_at = null;
        $question->save();

        return $question->fresh();
    }

    protected function notifyAdmin(Question $question): void
    {
        $admin = (string) (Settings::get('order_notify_email') ?: Settings::get('email'));

        if ($admin === '') {
            Log::warning('Elv Kit: некуда отправить письмо о новом вопросе', ['question' => $question->id]);

            return;
        }

        try {
            Mail::to($admin)->queue(new QuestionAdmin($question));
        }
        catch (\Throwable $e) {
            Log::warning('Elv Kit: письмо о новом вопросе не отправлено', [
                'question' => $question->id,
                'error' => $e->getMessage(),
            ]);
        }
    }
}

<?php namespace Elv\Kit\Classes;

use Elv\Kit\Classes\Mail\ReviewAdmin;
use Elv\Kit\Classes\Mail\ReviewInvite;
use Elv\Kit\Models\Order;
use Elv\Kit\Models\Product;
use Elv\Kit\Models\Review;
use Elv\Kit\Models\Settings;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\Mail;

/**
 * Reviews — отзывы покупателей, приглашения и модерация.
 *
 * Отзыв нельзя оставить с улицы: ссылка `review_token` выдаётся оплаченному
 * заказу и гасится после отправки. На витрину и в `AggregateRating` попадают
 * только одобренные отзывы.
 */
class Reviews
{
    public const INVITE_DAYS = 7;

    public static function enabled(): bool
    {
        return filter_var(Settings::get('reviews_enabled', true), FILTER_VALIDATE_BOOLEAN);
    }

    public static function inviteEnabled(): bool
    {
        return filter_var(Settings::get('review_invite_enabled', false), FILTER_VALIDATE_BOOLEAN);
    }

    public static function delayDays(): int
    {
        $days = (int) Settings::get('review_delay_days', self::INVITE_DAYS);

        return max(1, min(90, $days ?: self::INVITE_DAYS));
    }

    /**
     * approved — опубликованные отзывы продукта.
     *
     * @return \October\Rain\Database\Collection<int, Review>
     */
    public static function approved(Product $product, int $limit = 0)
    {
        $query = Review::query()
            ->approved()
            ->where('product_id', (int) $product->id);

        if ($limit > 0) {
            $query->limit($limit);
        }

        return $query->get();
    }

    /**
     * summary — количество и средняя оценка только по одобренным отзывам.
     *
     * @return array{count: int, average: float}
     */
    public static function summary(Product $product): array
    {
        $ratings = Review::query()
            ->where('product_id', (int) $product->id)
            ->where('status', Review::STATUS_APPROVED)
            ->pluck('rating')
            ->all();

        return [
            'count' => count($ratings),
            'average' => Community::average($ratings),
        ];
    }

    /**
     * aggregate — узел `AggregateRating` или null, если отзывов нет.
     *
     * @return array<string, mixed>|null
     */
    public static function aggregate(Product $product): ?array
    {
        $summary = static::summary($product);

        if ($summary['count'] < 1) {
            return null;
        }

        return [
            '@type' => 'AggregateRating',
            'ratingValue' => (string) $summary['average'],
            'reviewCount' => (string) $summary['count'],
            'bestRating' => '5',
            'worstRating' => '1',
        ];
    }

    /**
     * createInvitation — одноразовая ссылка для оплаченного заказа.
     */
    public function createInvitation(Order $order): string
    {
        $token = bin2hex(random_bytes(24));

        $order->forceFill([
            'review_token' => $token,
            'review_token_expires_at' => now()->addDays(30),
            'review_invited_at' => now(),
        ])->save();

        return $token;
    }

    public function invitationUrl(Order $order): string
    {
        return $order->review_token
            ? url('/review/' . $order->review_token) . '/'
            : '';
    }

    public function sendInvitation(Order $order): bool
    {
        $product = $order->product;

        if (!$product || trim((string) $order->email) === '') {
            return false;
        }

        $token = $this->createInvitation($order);

        try {
            Mail::to($order->email)->queue(new ReviewInvite($order, $product, $token));
        }
        catch (\Throwable $e) {
            Log::warning('Elv Kit: приглашение к отзыву не отправлено', [
                'order' => $order->order_no,
                'error' => $e->getMessage(),
            ]);

            return false;
        }

        return true;
    }

    /**
     * orderByToken — оплаченный заказ по действующей ссылке.
     */
    public function orderByToken(string $token): ?Order
    {
        $token = trim($token);

        if (preg_match('~^[a-f0-9]{48}$~', $token) !== 1) {
            return null;
        }

        $order = Order::query()
            ->where('review_token', $token)
            ->where('status', Order::STATUS_PAID)
            ->first();

        if (!$order || !$order->review_token_expires_at || $order->review_token_expires_at->isPast()) {
            return null;
        }

        return $order;
    }

    public function existsFor(Order $order): bool
    {
        if (!$order->product_id) {
            return false;
        }

        return Review::query()
            ->where('order_id', (int) $order->id)
            ->where('product_id', (int) $order->product_id)
            ->exists();
    }

    /**
     * submit — отзыв по ссылке из письма. Ждёт модерации.
     *
     * @param  array{author_name: string, rating: int, title?: string, text: string, pros?: string, cons?: string}  $data
     */
    public function submit(Order $order, array $data, string $ip = ''): ?Review
    {
        if (!$order->product_id || $this->existsFor($order)) {
            return null;
        }

        $review = new Review();
        $review->fill([
            'product_id' => (int) $order->product_id,
            'order_id' => (int) $order->id,
            'author_name' => (string) $data['author_name'],
            'email_hash' => Community::emailHash((string) $order->email),
            'rating' => max(1, min(5, (int) $data['rating'])),
            'title' => (string) ($data['title'] ?? ''),
            'text' => (string) $data['text'],
            'pros' => (string) ($data['pros'] ?? ''),
            'cons' => (string) ($data['cons'] ?? ''),
            'status' => Review::STATUS_PENDING,
            'is_verified' => true,
            'ip_hash' => Community::ipHash($ip),
        ]);
        $review->save();

        $order->forceFill([
            'review_token' => null,
            'review_token_expires_at' => null,
        ])->save();

        $this->notifyAdmin($review);

        return $review;
    }

    public function approve(Review $review): Review
    {
        $review->status = Review::STATUS_APPROVED;
        $review->published_at = $review->published_at ?: now();
        $review->save();

        return $review->fresh();
    }

    public function reject(Review $review): Review
    {
        $review->status = Review::STATUS_REJECTED;
        $review->published_at = null;
        $review->save();

        return $review->fresh();
    }

    protected function notifyAdmin(Review $review): void
    {
        $admin = (string) (Settings::get('order_notify_email') ?: Settings::get('email'));

        if ($admin === '') {
            Log::warning('Elv Kit: некуда отправить письмо о новом отзыве', ['review' => $review->id]);

            return;
        }

        try {
            Mail::to($admin)->queue(new ReviewAdmin($review));
        }
        catch (\Throwable $e) {
            Log::warning('Elv Kit: письмо о новом отзыве не отправлено', [
                'review' => $review->id,
                'error' => $e->getMessage(),
            ]);
        }
    }
}

<?php namespace Elv\Kit\Classes\Http;

use Elv\Kit\Classes\AntiSpam;
use Elv\Kit\Classes\Catalog;
use Elv\Kit\Classes\Questions;
use Elv\Kit\Models\Product;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;

/**
 * QuestionController — вопрос к продукту со страницы витрины.
 *
 * Форма открыта без входа в кабинет: вопрос покупателя не требует заказа.
 * Запись создаётся в статусе `new` и на витрине не появляется, пока её не
 * опубликуют ответом. Возврат — на ту же страницу продукта к блоку «Вопрос-ответ».
 */
class QuestionController extends FormController
{
    protected ?Product $product = null;

    public function __construct(protected Questions $questions = new Questions())
    {
    }

    protected function rules(): array
    {
        return [
            'product' => 'required|string|max:160',
            'name' => 'required|string|max:160',
            'email' => 'required|email|max:191',
            'question' => 'required|string|min:10|max:3000',
            'consent' => 'accepted',
        ];
    }

    protected function messages(): array
    {
        return [
            'product.required' => 'Не выбран продукт для вопроса.',
            'name.required' => 'Укажите имя.',
            'email.required' => 'Укажите e-mail — на него ответим.',
            'email.email' => 'Проверьте адрес e-mail.',
            'question.required' => 'Напишите вопрос.',
            'question.min' => 'Опишите вопрос подробнее — минимум 10 символов.',
            'consent.accepted' => 'Нужно согласие на обработку персональных данных.',
        ];
    }

    /**
     * @param  array<string, mixed>  $data
     * @return JsonResponse|RedirectResponse|null
     */
    protected function handle(array $data, Request $request)
    {
        if (!Questions::enabled()) {
            return $this->invalid($request, ['question' => ['Приём вопросов временно закрыт.']]);
        }

        $this->product = Catalog::find((string) ($data['product'] ?? ''));

        if (!$this->product) {
            return $this->invalid($request, ['product' => ['Такой продукт не найден.']]);
        }

        $this->questions->ask($this->product, [
            'name' => (string) $data['name'],
            'email' => (string) $data['email'],
            'question' => (string) $data['question'],
        ], AntiSpam::ip($request));

        return null;
    }

    protected function fallbackPath(): string
    {
        if ($this->product) {
            return $this->product->kind === 'kit'
                ? '/kit/'
                : '/modules/' . $this->product->slug . '/';
        }

        return '/kit/';
    }

    protected function anchor(): string
    {
        return 'qa';
    }

    protected function done(Request $request, array $extra = [])
    {
        if ($request->expectsJson()) {
            return response()->json(array_merge(['ok' => true], $extra), 201);
        }

        return $this->back($request, ['sent' => '1'])
            ->with('elv_success', 'Вопрос отправлен. Ответ появится на этой странице после проверки.');
    }
}

<?php namespace Elv\Kit\Classes\Http;

use Elv\Kit\Classes\AntiSpam;
use Elv\Kit\Classes\Reviews;
use Elv\Kit\Models\Order;
use Elv\Kit\Models\Product;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;

/**
 * ReviewController — отзыв по одноразовой ссылке из письма.
 *
 * Ссылка выдаётся оплаченному заказу (`review_token`) и гасится после отправки.
 * Отзыв ждёт модерации: на витрину и в `AggregateRating` попадают только
 * одобренные. Просроченная или использованная ссылка отвечает 404.
 */
class ReviewController extends FormController
{
    protected ?Order $order = null;

    protected ?Product $product = null;

    protected string $token = '';

    public function __construct(protected Reviews $reviews = new Reviews())
    {
    }

    protected function rules(): array
    {
        return [
            'author_name' => 'required|string|max:160',
            'rating' => 'required|integer|min:1|max:5',
            'title' => 'nullable|string|max:160',
            'text' => 'required|string|min:20|max:4000',
            'pros' => 'nullable|string|max:1000',
            'cons' => 'nullable|string|max:1000',
            'consent' => 'accepted',
        ];
    }

    protected function messages(): array
    {
        return [
            'author_name.required' => 'Укажите имя.',
            'rating.required' => 'Поставьте оценку от 1 до 5.',
            'rating.min' => 'Оценка — от 1 до 5.',
            'rating.max' => 'Оценка — от 1 до 5.',
            'text.required' => 'Напишите отзыв.',
            'text.min' => 'Расскажите подробнее — минимум 20 символов.',
            'consent.accepted' => 'Нужно согласие на публикацию отзыва.',
        ];
    }

    /**
     * @param  array<string, mixed>  $data
     * @return JsonResponse|RedirectResponse|null
     */
    protected function handle(array $data, Request $request)
    {
        if (!Reviews::enabled()) {
            abort(404);
        }

        $this->token = (string) $request->route('token');
        $this->order = $this->reviews->orderByToken($this->token);

        if (!$this->order) {
            abort(404);
        }

        $this->product = $this->order->product;

        if ($this->reviews->existsFor($this->order)) {
            return $this->invalid($request, ['text' => ['Отзыв по этому заказу уже отправлен.']]);
        }

        $this->reviews->submit($this->order, [
            'author_name' => (string) $data['author_name'],
            'rating' => (int) $data['rating'],
            'title' => (string) ($data['title'] ?? ''),
            'text' => (string) $data['text'],
            'pros' => (string) ($data['pros'] ?? ''),
            'cons' => (string) ($data['cons'] ?? ''),
        ], AntiSpam::ip($request));

        return null;
    }

    protected function fallbackPath(): string
    {
        return '/review/' . $this->token . '/';
    }

    protected function anchor(): string
    {
        return 'review-form';
    }

    protected function done(Request $request, array $extra = [])
    {
        if ($request->expectsJson()) {
            return response()->json(array_merge(['ok' => true], $extra), 201);
        }

        // Токен гасится после отправки, поэтому возврат — на страницу продукта:
        // ссылка на отзыв к этому моменту уже недействительна.
        $url = $this->product ? (string) $this->product->url : rtrim(url('/'), '/') . '/';

        return redirect($url . '?review=sent#reviews')
            ->with('elv_success', 'Спасибо! Отзыв отправлен на проверку — опубликуем после модерации.');
    }
}

 * Question — вопрос к продукту.

 * Отправитель создаёт строку в статусе `new`. На витрину она попадает только
 * после ответа и модерации: `published` ставит админка, а `answer` не пустой.
 */
class Question extends Model
{
    public const STATUS_NEW = 'new';
    public const STATUS_PUBLISHED = 'published';
    public const STATUS_REJECTED = 'rejected';

    public $table = 'elv_kit_questions';

    public $fillable = [
        'product_id', 'customer_id', 'name', 'email', 'question', 'answer',
        'answered_by', 'status', 'is_pinned', 'sort_order', 'answered_at',
        'published_at', 'ip_hash',
    ];

    public $rules = [
        'product_id' => 'required|integer|min:1',
        'name' => 'required|max:160',
        'email' => 'required|email|max:191',
        'question' => 'required|max:3000',
        'status' => 'required|in:new,published,rejected',
    ];

    protected $casts = [
        'is_pinned' => 'boolean',
        'answered_at' => 'datetime',
        'published_at' => 'datetime',
    ];

    public $belongsTo = [
        'product' => Product::class,
        'customer' => Customer::class,
    ];

    public function scopePublished($query)
    {
        return $query
            ->where('status', self::STATUS_PUBLISHED)
            ->whereNotNull('answer')
            ->where('answer', '<>', '')
            ->orderByDesc('is_pinned')
            ->orderBy('sort_order')
            ->orderByDesc('published_at');
    }

    public function getDisplayNameAttribute(): string
    {
        return Community::displayName((string) $this->name);
    }

    public function getStatusLabelAttribute(): string
    {
        return match ((string) $this->status) {
            self::STATUS_PUBLISHED => 'Опубликован',
            self::STATUS_REJECTED => 'Отклонён',
            default => 'Новый',
        };
    }

    public function getIsPublishedAttribute(): bool
    {
        return (string) $this->status === self::STATUS_PUBLISHED && trim((string) $this->answer) !== '';
    }
}

use Model;

/**
 * Review — отзыв покупателя с оценкой 1–5.
 *
 * Отзыв приходит по одноразовой ссылке из письма и ждёт модерации. На витрину

class Review extends Model

    public const STATUS_PENDING = 'pending';
    public const STATUS_APPROVED = 'approved';
    public const STATUS_REJECTED = 'rejected';

    public $table = 'elv_kit_reviews';

        'product_id', 'order_id', 'customer_id', 'author_name', 'email_hash',
        'rating', 'title', 'text', 'pros', 'cons', 'status', 'is_verified',

    ];

    public $rules = [
        'product_id' => 'required|integer|min:1',
        'author_name' => 'required|max:160',
        'rating' => 'required|integer|min:1|max:5',
        'text' => 'required|max:4000',
        'status' => 'required|in:pending,approved,rejected',
    ];

    protected $casts = [
        'is_verified' => 'boolean',
        'replied_at' => 'datetime',
        'published_at' => 'datetime',
    ];

    public $belongsTo = [
        'product' => Product::class,
        'order' => Order::class,
        'customer' => Customer::class,
    ];

    public function scopeApproved($query)
    {
        return $query
            ->where('status', self::STATUS_APPROVED)
            ->orderByDesc('published_at')
            ->orderByDesc('id');
    }

    public function getDisplayNameAttribute(): string
    {
        return Community::displayName((string) $this->author_name);
    }

    public function getStatusLabelAttribute(): string
    {
        return match ((string) $this->status) {
            self::STATUS_APPROVED => 'Одобрен',
            self::STATUS_REJECTED => 'Отклонён',
            default => 'На модерации',
        };
    }

    public function getIsApprovedAttribute(): bool
    {
        return (string) $this->status === self::STATUS_APPROVED;
    }

    /**
     * stars — пять флагов для шаблона: `true` для заполненной звезды.
     *
     * @return array<int, bool>
     */
    public function getStarsAttribute(): array
    {
        $rating = max(0, min(5, (int) $this->rating));

        return array_map(fn (int $star) => $star <= $rating, range(1, 5));
    }
}

$check($reviews->orderByToken($token) === null, 'expired token still finds the order');
$order->forceFill(['review_token_expires_at' => now()->addDays(30)])->save();

$unpaid = Order::create([
    'order_no' => Order::nextNumber(),
    'product_id' => (int) $product->id,
    'product_name' => (string) $product->name,
    'plan' => 'single',
    'qty' => 1,
    'amount_cents' => 1000,
    'currency' => 'BYN',
    'status' => Order::STATUS_PENDING,
    'customer_name' => 'Мария Иванова',
    'email' => $email,
]);

$unpaidToken = $reviews->createInvitation($unpaid);
$check($reviews->orderByToken($unpaidToken) === null, 'unpaid order is found by token');

$review = $reviews->submit($order->fresh(), [
    'author_name' => 'Алексей Кузнецов',
    'rating' => 5,
    'title' => 'Запустили за вечер',
    'text' => 'Установили на свой домен, запись и оплата работают. Поддержка ответила в тот же день.',
    'pros' => 'Готовые страницы',
    'cons' => 'Пришлось разобраться с почтой',
], '127.0.0.1');

$check($review instanceof Review, 'review is not created');
$check((string) $review->status === Review::STATUS_PENDING, 'new review is not on moderation');
$check($review->is_verified === true, 'review from a paid order is not verified');
$check($review->display_name === 'Алексей К.', 'review display name is not «Имя Ф.»');
$check($review->stars === [true, true, true, true, true], 'five-star review has the wrong stars');
$check($order->fresh()->review_token === null, 'review token is not cleared after submit');
$check($reviews->existsFor($order->fresh()) === true, 'submitted review is not found for the order');
$check($reviews->submit($order->fresh(), [
    'author_name' => 'Алексей Кузнецов',
    'rating' => 5,
    'text' => 'Повторный отзыв по тому же заказу.',
]) === null, 'second review for the same order is accepted');

$check(Reviews::summary($product)['count'] === 0, 'pending review is already counted');
$check(Reviews::approved($product)->count() === 0, 'pending review is already published');
$check(Reviews::aggregate($product) === null, 'aggregate rating exists without approved reviews');

$reviews->approve($review);
$review = $review->fresh();

$check((string) $review->status === Review::STATUS_APPROVED, 'approve does not change the status');
$check($review->published_at !== null, 'approved review has no publish date');
$check(Reviews::summary($product)['count'] === 1, 'approved review is not counted');
$check(Reviews::summary($product)['average'] === 5.0, 'approved review average is not 5.0');
$check(Reviews::approved($product)->count() === 1, 'approved review is not listed');

$aggregate = Reviews::aggregate($product);
$check($aggregate !== null, 'aggregate rating is missing with an approved review');
$check(($aggregate['ratingValue'] ?? '') === '5', 'aggregate rating value is not 5');
$check(($aggregate['reviewCount'] ?? '') === '1', 'aggregate rating count is not 1');

$json = Seo::jsonld($page);
